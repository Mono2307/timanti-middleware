// Label + note + gray-out state for NEWDRAFT_CELL, in ONE idempotent place. Called by the fresh
// build, the in-place migration, the one-time doc-type setup and the standalone menu item, so an
// existing live sheet can be brought up to date WITHOUT a destructive rebuild (a rebuild clears
// every staff input on the tab).
//
// The gray-out rule is REMOVED here, never re-added. Earlier versions grayed this cell whenever
// Document Type = Voucher, because only the Exchange Note read it — and a conditional format
// SURVIVES a rebuild (the builders clear content and validations, not format rules), so without
// this the cell would keep reading as "don't fill this in" long after vouchers could use it.
function refreshNewDraftField_(calc) {
  var r = calc.getRange(NEWDRAFT_CELL);
  calc.getRange(r.getRow(), r.getColumn() - 1).setValue(NEWDRAFT_LABEL);
  r.setNote(NEWDRAFT_NOTE);
  calc.setConditionalFormatRules(calc.getConditionalFormatRules().filter(function (rule) {
    return !rule.getRanges().some(function (g) { return g.getA1Notation() === r.getA1Notation(); });
  }));
}

// Menu-visible wrapper: brings an EXISTING sheet up to date after the script is re-pasted, without

// One-time structural setup: builds the Document Type dropdown (DOCTYPE_CELL) + New Draft/Order #
// field (NEWDRAFT_CELL), labels, default and help note.
// Safe to re-run (idempotent). Also creates the Voucher Log / Exchange Log tabs if missing.
// NOTE: this used to gray the New Draft cell out whenever "Voucher" was selected, because only the
// Exchange Note read it. A voucher can now be applied to a draft too, so the cell is live for both
// document types and any leftover gray-out rule is removed on re-run.
function setupDocTypeFields() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  var docTypeRange  = calc.getRange(DOCTYPE_CELL);
  var newDraftRange = calc.getRange(NEWDRAFT_CELL);

  // SAFETY GUARD: refuse to write if the target cells (or their labels) already hold
  // something that isn't ours. Protects the live sheet even if the row guess is off.
  var docLabelRange = calc.getRange(docTypeRange.getRow(),  docTypeRange.getColumn()  - 1);
  var drfLabelRange = calc.getRange(newDraftRange.getRow(), newDraftRange.getColumn() - 1);
  var ours = { 'Document Type': 1, 'Voucher': 1, 'Exchange Note': 1 };
  ours[NEWDRAFT_LABEL] = 1;
  ours[NEWDRAFT_LABEL_LEGACY] = 1;   // sheets built before the voucher could be applied to a draft
  var blocked = [docLabelRange, docTypeRange, drfLabelRange, newDraftRange].filter(function (r) {
    var v = String(r.getValue()).trim();
    return v !== '' && !ours[v];
  });
  if (blocked.length) {
    ui.alert('Aborted — row ' + docTypeRange.getRow() + ' is not empty.\n\n' +
      blocked.map(function (r) { return '  • ' + r.getA1Notation() + ' = "' + String(r.getValue()).trim() + '"'; }).join('\n') +
      '\n\nNothing was changed. Tell me a different free row and I\'ll move DOCTYPE_CELL / NEWDRAFT_CELL.');
    return;
  }

  // Labels in the column immediately left of each field. (NEWDRAFT_CELL's label, note and gray-out
  // state are owned by refreshNewDraftField_, called below.)
  calc.getRange(docTypeRange.getRow(), docTypeRange.getColumn() - 1).setValue('Document Type');

  // Dropdown on the Document Type cell.
  var rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['Voucher', 'Exchange Note'], true)
    .setAllowInvalid(false)
    .setHelpText('Voucher = 1-year store credit (discount code). Exchange Note = instant deduction on a new invoice.')
    .build();
  docTypeRange.setDataValidation(rule);
  if (!String(docTypeRange.getValue()).trim()) docTypeRange.setValue('Voucher');

  refreshNewDraftField_(calc);

  ensureLogTabs_(ss);

  ui.alert('Document Type fields ready:\n\n' +
    '• ' + DOCTYPE_CELL + ' — dropdown (Voucher / Exchange Note), default Voucher\n' +
    '• ' + NEWDRAFT_CELL + ' — New Draft/Order # (required for Exchange Note, optional for Voucher)\n\n' +
    'Tabs: "' + VOUCHER_LOG + '", "' + EXCHANGE_LOG + '" and "' + AFTERSALES_LOG + '" are present.');
}

// ── AUTO-FILL: INSTALLABLE onEdit HANDLER ────────────────────────────────────
// Installed via setupTriggers(). Runs under the account that created the trigger.
// Errors are surfaced via ui.alert so silent failures can't happen.
function handleEdit(e) {
  try {
    const sheet = e.range.getSheet();
    if (sheet.getName() !== CALC_SHEET_NAME) return;
    const col = e.range.getColumn();
    const row = e.range.getRow();
    const a1 = e.range.getA1Notation();
    if (col === 2 && row === cellRow_(ORDER_NUM_CELL)) { onOrderNumberEntered(sheet); try { refreshFinalIssueValue_(sheet); } catch (x) {} return; }
    if (col === 2 && row === cellRow_(SKU_CELL))       { onSkuSelected(sheet);        return; }
    // Flipping Source or Exchange Type updates the "Value to Issue" cell live.
    if (a1 === SOURCE_CELL || a1 === EXCHTYPE_CELL) { try { refreshFinalIssueValue_(sheet); } catch (x) {} return; }
    // Old-gold auto-fill: purity → buy-back rate; weight → value; customer phone/email → resolve.
    if (a1 === OG_PURITY_CELL) { try { fetchOldGoldRate(); refreshFinalIssueValue_(sheet); } catch (x) {} return; }
    if (a1 === OG_WEIGHT_CELL) { try { showOldGoldValue_(sheet, oldGoldValue_(sheet)); refreshFinalIssueValue_(sheet); } catch (x) {} return; }
    if (a1 === OG_CUSTOMER_CELL) { try { lookupOldGoldCustomer(); } catch (x) {} return; }
  } catch (err) {
    SpreadsheetApp.getUi().alert('❌ Auto-fill error:\n' + err.message);
  }
}

// ── ORDER NUMBER → populate customer fields + SKU picker ─────────────────────
// allowModal: true when called from a menu item (has container.ui scope);
//             false/omitted when called from the installable onEdit trigger (no UI scope).
function onOrderNumberEntered(sheet, allowModal) {
  const raw = String(sheet.getRange(ORDER_NUM_CELL).getValue()).trim();
  if (!raw) return;

  // Clear all previously auto-filled cells
  [CUST_NAME_CELL, CUST_EMAIL_CELL, CUST_PHONE_CELL, ORDER_DATE_CELL, SKU_CELL, KARAT_CELL,
   NET_WT_CELL, DIA_CTS_CELL, GOLD_RATE_ORD_CELL, LGD_RATE_CELL,
   DIA_VALUE_CELL, DIA_PAID_CELL, GOLD_RATE_EFF_CELL, GOLD_VAL_CELL, DIA_VAL_CELL]
   // Both sides of the before/after table. A paid figure left over from the previous order would sit
   // under the new one looking like its history — the single worst failure this table could have.
   .concat(CALC_ROWS.map(function (r) { return r.paid; }))
   .concat(CALC_ROWS.map(function (r) { return r.value; }))
   .forEach(function(ref) {
    sheet.getRange(ref).clearContent();
  });
  // The rate cell carries the per-SKU breakdown in its note; a stale note under a new order
  // would read as this order's rates.
  sheet.getRange(GOLD_RATE_EFF_CELL).clearNote();
  sheet.getRange(SKU_CELL).clearDataValidations();

  const orderName = raw.replace('#', '');

  // Single API call — fetch everything needed in one shot
  const data = shopifyGet(
    'orders.json?name=%23' + orderName +
    '&status=any&fields=id,customer,line_items,shipping_address,billing_address,created_at'
  );

  if (!data || !data.orders || data.orders.length === 0) {
    SpreadsheetApp.getUi().alert('Order #' + orderName + ' not found in Shopify.');
    return;
  }

  const order    = data.orders[0];
  const customer = order.customer || {};

  // ── Customer fields ──────────────────────────────────────────────────────
  const name  = [customer.first_name, customer.last_name].filter(Boolean).join(' ');
  const email = customer.email || '';

  // Phone: shipping first, fallback to billing
  var phone = '';
  if (order.shipping_address && order.shipping_address.phone) {
    phone = order.shipping_address.phone;
  } else if (order.billing_address && order.billing_address.phone) {
    phone = order.billing_address.phone;
  }

  // Order date — guard against missing/invalid created_at
  var orderDate = '';
  try {
    if (order.created_at) {
      orderDate = Utilities.formatDate(new Date(order.created_at), 'Asia/Kolkata', 'dd-MM-yyyy');
    }
  } catch (_) {}

  sheet.getRange(CUST_NAME_CELL).setValue(name);
  sheet.getRange(CUST_EMAIL_CELL).setValue(email);
  sheet.getRange(CUST_PHONE_CELL).setValue(phone);
  sheet.getRange(ORDER_DATE_CELL).setValue(orderDate);

  // ── Line items ───────────────────────────────────────────────────────────
  const lineItems = order.line_items || [];
  if (lineItems.length === 0) return;

  // Count UNITS, not lines — a single qty-2 line must still offer a picker, or staff would
  // silently exchange one unit's worth of gold for a two-unit purchase.
  const expanded = expandLineItems_(lineItems);

  if (expanded.length === 1) {
    // Auto-select and populate immediately
    sheet.getRange(SKU_CELL).setValue(expanded[0].label);
    populateFromLineItem(sheet, lineItems[0]);
    return;
  }

  // Multiple SKUs — two paths depending on whether we have UI access:
  //   Menu ("Lookup Order Now") → modal dialog, supports multi-select + aggregation
  //   Trigger (auto onEdit)    → dropdown on B10, single-SKU selection
  if (allowModal) {
    showSkuCheckboxDialog(lineItems);
    return;
  }

  // Trigger path: no container.ui scope — use data validation dropdown instead.
  // User picks one SKU; onSkuSelected fires and populates that item's fields.
  // For multi-SKU aggregation use "🔄 Lookup Order Now" from the menu.
  var skuList = expanded.map(function(e) { return e.label; });
  var rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(skuList, true)
    .setAllowInvalid(false)
    .build();
  sheet.getRange(SKU_CELL).clearContent();
  sheet.getRange(SKU_CELL).setDataValidation(rule);
  sheet.getRange(SKU_CELL).setNote(
    skuList.length + ' SKUs found.\n' +
    '• Pick one below for single-item rates.\n' +
    '• Use "🔄 Lookup Order Now" in the menu to select multiple SKUs.'
  );
}

// Expands each Shopify line item into one entry PER UNIT. Shopify stores "2 of the same ring" as a
// single line item with quantity:2 — the old code counted it once, paying out half the gold. Each
// unit becomes its own pickable row ("SKU (1 of 2)") so staff exchange exactly the units in hand.
// Returns [{ li, label }]; li is shared by reference (read-only downstream).
function expandLineItems_(lineItems) {
  var out = [];
  (lineItems || []).forEach(function (li) {
    var qty  = Math.max(1, parseInt(li.quantity, 10) || 1);
    var base = li.sku || li.title || '';
    for (var n = 1; n <= qty; n++) {
      out.push({ li: li, label: qty > 1 ? base + ' (' + n + ' of ' + qty + ')' : base });
    }
  });
  return out;
}

// Strips the " (n of q)" unit suffix added by expandLineItems_, recovering the raw SKU.
function stripUnitSuffix_(label) {
  return String(label || '').replace(/\s*\(\d+ of \d+\)\s*$/, '').trim();
}

// ── MULTI-SKU PROMPT (menu path — avoids showModalDialog scope restriction) ───
// Uses ui.prompt (plain text input) which works with the same scope as alert().
function showSkuCheckboxDialog(lineItems) {
  var ui       = SpreadsheetApp.getUi();
  // One row per UNIT, not per line item — a qty-2 line offers two tickable rows.
  var expanded = expandLineItems_(lineItems);
  var numbered = expanded.map(function(e, i) {
    return (i + 1) + '. ' + (e.label || 'Item ' + (i + 1));
  }).join('\n');

  var result = ui.prompt(
    'Select SKUs for Exchange',
    numbered + '\n\nEnter item numbers separated by commas (e.g. "1" or "1,2"):',
    ui.ButtonSet.OK_CANCEL
  );

  if (result.getSelectedButton() !== ui.Button.OK) return;

  var raw     = result.getResponseText().trim();
  var indices = raw.split(/[,\s]+/)
    .map(function(s) { return parseInt(s.trim(), 10) - 1; })
    .filter(function(i) { return !isNaN(i) && i >= 0 && i < expanded.length; });

  if (!indices.length) {
    ui.alert('No valid numbers entered. Use "1" for item 1, "1,2" for both.');
    return;
  }

  // Store so applySkuSelection can look them up
  PropertiesService.getScriptProperties().setProperty('_PENDING_LINE_ITEMS', JSON.stringify(lineItems));
  applySkuSelection(indices);
}

// Called directly from showSkuCheckboxDialog (indices already resolved)
function applySkuSelection(selectedIndices) {
  var json = PropertiesService.getScriptProperties().getProperty('_PENDING_LINE_ITEMS');
  if (!json) throw new Error('Session expired — re-enter the order number and try again.');

  // Rebuild the per-unit expansion rather than storing it — duplicated line-item objects would
  // push the stored JSON toward the 9 KB Script Properties limit that already bit this flow once.
  var lineItems = JSON.parse(json);
  var expanded  = expandLineItems_(lineItems);
  var selected  = selectedIndices.map(function(i) { return expanded[i].li; });

  var sheet    = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CALC_SHEET_NAME);
  var skuLabel = selectedIndices.map(function(i) { return expanded[i].label; }).join(', ');
  sheet.getRange(SKU_CELL).setValue(skuLabel);
  sheet.getRange(SKU_CELL).clearDataValidations();
  sheet.getRange(SKU_CELL).clearNote();

  if (selected.length === 1) {
    populateFromLineItem(sheet, selected[0]);
  } else {
    populateFromMultipleLineItems(sheet, selected);
  }
}

// Aggregate net wt, dia cts, gold value, and dia value across multiple SKUs.
// GOLD_RATE_ORD_CELL shows the order-time rates as "X / Y" (human-readable only).
//
// GOLD_RATE_EFF_CELL carries the WEIGHTED EFFECTIVE gold rate (total gold value ÷ total weight) so
// GOLD_VAL_CELL can stay a real formula in both the single- and multi-SKU cases. Previously the
// multi case wrote the gold value as a static number — arithmetically right at the instant of
// selection, but frozen: editing the weight afterwards left the value stale with no warning.
// It must stay NUMERIC for that reason, so the per-SKU rates go in its note, not in the cell.
function populateFromMultipleLineItems(sheet, lineItems) {
  var totalNetWt       = 0;
  var totalDiaCts      = 0;
  var totalLiveDiaVal  = 0;
  var totalLiveGoldVal = 0;
  var orderRates       = [];
  var liveRates        = [];
  var karat            = null;
  var karats           = [];   // every distinct karat seen — a mixed lot must not silently show one
  var ratedNetWt       = 0;    // weight of items that actually had a live rate (weighted-avg base)
  var hasNetWt         = false;
  var hasDiaCts        = false;
  var hasDiaVal        = false;
  // Paid-side totals across the selected items. Tracked with a "did we see one" flag apiece so a lot
  // where NO item recorded a component leaves that cell blank instead of claiming a charge of zero.
  var paid = { gold: 0, dia: 0, gem: 0, making: 0, disc: 0, tax: 0, diaPaid: 0 };
  var sawPaid = { gold: false, dia: false, gem: false, making: false, disc: false, tax: false, diaPaid: false };

  lineItems.forEach(function(li) {
    var props = {};
    (li.properties || []).forEach(function(p) { props[p.name] = p.value; });

    var netWt         = props['_net_wt']      != null ? parseFloat(props['_net_wt'])      : null;
    var diaCts        = props['_diamond_cts'] != null ? parseFloat(props['_diamond_cts']) : null;
    var goldRateOrder = props['_gold_rate']   != null ? parseFloat(props['_gold_rate'])   : null;
    var goldRateLive  = null;
    var diaValue      = null;

    if (li.variant_id) {
      var vmData = shopifyGet('variants/' + li.variant_id + '/metafields.json');
      if (vmData && vmData.metafields) {
        var vmf = {};
        vmData.metafields.forEach(function(mf) {
          if (mf.namespace === 'custom') vmf[mf.key] = mf.value;
        });
        if (netWt === null && vmf['net_metal_weight_g']   != null) netWt        = parseFloat(vmf['net_metal_weight_g']);
        if (vmf['gold_rate']             != null)                   goldRateLive = parseFloat(vmf['gold_rate']);
        if (vmf['price_breakup_diamond'] != null)                   diaValue     = parseFloat(vmf['price_breakup_diamond']);
      }
    }

    if (diaCts === null && li.product_id) {
      var pmData = shopifyGet('products/' + li.product_id + '/metafields.json');
      if (pmData && pmData.metafields) {
        pmData.metafields.forEach(function(mf) {
          if (mf.namespace === 'custom' && mf.key === 'totaldiamondweight' && diaCts === null) {
            diaCts = parseFloat(mf.value);
          }
        });
      }
    }

    if (netWt        !== null) { totalNetWt  += netWt;  hasNetWt  = true; }
    if (diaCts       !== null) { totalDiaCts += diaCts; hasDiaCts = true; }
    if (diaValue     !== null) { totalLiveDiaVal += diaValue; hasDiaVal = true; }
    if (goldRateOrder !== null) orderRates.push(goldRateOrder);
    if (goldRateLive  !== null) {
      liveRates.push(goldRateLive);
      if (netWt !== null) { totalLiveGoldVal += netWt * goldRateLive; ratedNetWt += netWt; }
    }
    // Paid side — the Rs-prefixed breakup properties frozen on the line at purchase.
    var pc = paidComponents_(props, li);
    ['gold', 'dia', 'making', 'disc', 'tax', 'diaPaid'].forEach(function (k) {
      if (pc[k] !== null && pc[k] !== undefined) { paid[k] += pc[k]; sawPaid[k] = true; }
    });

    var k = extractKarat(li.sku);
    if (k && karats.indexOf(k) === -1) karats.push(k);
    if (!karat) karat = k;
  });

  if (hasNetWt)  sheet.getRange(NET_WT_CELL).setValue(totalNetWt);
  if (hasDiaCts) sheet.getRange(DIA_CTS_CELL).setValue(totalDiaCts);
  // Mixed-karat lots show every karat ("18K / 22K") rather than silently printing the first one.
  if (karats.length) sheet.getRange(KARAT_CELL).setValue(karats.join(' / '));

  if (orderRates.length) sheet.getRange(GOLD_RATE_ORD_CELL).setValue(orderRates.join(' / '));

  if (hasDiaVal) {
    sheet.getRange(LGD_RATE_CELL).setValue(totalLiveDiaVal);
    sheet.getRange(DIA_VALUE_CELL).setValue(totalLiveDiaVal);
  }

  // Weighted effective rate → B27 stays a live formula. Divide by the RATED weight only, so an
  // item with no live rate can't dilute the average; if that leaves rated < total weight the
  // formula would over-credit the unrated grams, so warn instead of silently mispricing.
  if (totalLiveGoldVal > 0 && ratedNetWt > 0) {
    sheet.getRange(GOLD_RATE_EFF_CELL).setValue(totalLiveGoldVal / ratedNetWt);
    sheet.getRange(GOLD_RATE_EFF_CELL).setNote(goldRateNote_(true, liveRates));
    sheet.getRange(GOLD_VAL_CELL).setFormula('=' + NET_WT_CELL + '*' + GOLD_RATE_EFF_CELL);
    if (Math.abs(ratedNetWt - totalNetWt) > 0.0001) {
      SpreadsheetApp.getUi().alert(
        '⚠️ Some selected items have no live gold rate.\n\n' +
        'Rated weight: ' + ratedNetWt.toFixed(3) + ' g of ' + totalNetWt.toFixed(3) + ' g total.\n\n' +
        'The gold value now applies the weighted rate to the FULL weight, which over-credits the ' +
        'unrated grams. Check the variant metafields before issuing.');
    }
  } else if (liveRates.length) {
    // Rates were read but no weight to average them over, so there is no effective rate to
    // multiply. Show what was read rather than leaving the cell blank — a blank rate under a
    // populated order reads as "Shopify had nothing", which is a different problem.
    sheet.getRange(GOLD_RATE_EFF_CELL).setValue(liveRates.join(' / '));
    sheet.getRange(GOLD_RATE_EFF_CELL).setNote(
      'Live gold rates read from the variants, but no net weight to weight them by — so there is ' +
      'no effective rate and the gold value cannot be computed. Check custom.net_metal_weight_g.');
  }
  if (hasDiaVal) sheet.getRange(DIA_VAL_CELL).setFormula('=' + DIA_VALUE_CELL + '*0.8');

  var r2 = function (n) { return Math.round(n * 100) / 100; };
  writePaidColumn_(sheet, {}, { totals: {
    gold:   sawPaid.gold   ? r2(paid.gold)   : null,
    dia:    sawPaid.dia    ? r2(paid.dia)    : null,
    gem:    null,                     // no gemstone value exists in the system — see paidComponents_
    making: sawPaid.making ? r2(paid.making) : null,
    ship:   null,
    igi:    null,
    disc:   sawPaid.disc    ? r2(paid.disc)    : null,
    tax:    sawPaid.tax     ? r2(paid.tax)     : null,
    // Capping a multi-SKU lot on the SUM of what was paid for its stones is the same test applied
    // to the lot as a whole, which is the level the credit is issued at.
    diaPaid: sawPaid.diaPaid ? r2(paid.diaPaid) : null
  }});

  SpreadsheetApp.flush();
}

// ── SKU SELECTED → populate jewel fields ─────────────────────────────────────
// Does a fresh Shopify call — avoids Script Properties 9 KB limit that caused
// silent cache failures. Low-volume tool so the extra call is fine.
function onSkuSelected(sheet) {
  const selected  = String(sheet.getRange(SKU_CELL).getValue()).trim();
  if (!selected) return;

  const orderName = String(sheet.getRange(ORDER_NUM_CELL).getValue()).trim().replace('#', '');
  if (!orderName) return;

  const data = shopifyGet(
    'orders.json?name=%23' + orderName + '&status=any&fields=id,line_items'
  );

  if (!data || !data.orders || data.orders.length === 0) {
    SpreadsheetApp.getUi().alert('Order #' + orderName + ' not found — cannot load SKU data.');
    return;
  }

  const lineItems = data.orders[0].line_items || [];
  const wanted    = stripUnitSuffix_(selected);   // dropdown labels carry a " (1 of 2)" unit suffix
  const li        = lineItems.find(function(item) {
    return (item.sku || item.title) === wanted;
  });

  if (!li) {
    SpreadsheetApp.getUi().alert('SKU "' + selected + '" not found in order #' + orderName + '.');
    return;
  }

  populateFromLineItem(sheet, li);
}

// ── POPULATE JEWEL FIELDS FROM A SINGLE LINE ITEM ────────────────────────────
// B33 = gold rate locked at order time (line item property _gold_rate)
// C33 = effective gold rate now        (variant metafield custom.gold_rate) — the ONE rate cell
// B34 = live diamond value             (variant metafield custom.price_breakup_diamond)
// C34 = same live diamond value        (shown in C column for formula use)
// B41 = live gold value formula  → =B29*C33  (adjust if your formula differs)
// B42 = live diamond value formula → =C34*0.8
function populateFromLineItem(sheet, lineItem) {
  var props = {};
  (lineItem.properties || []).forEach(function(p) { props[p.name] = p.value; });

  var netWt         = props['_net_wt']      != null ? parseFloat(props['_net_wt'])      : null;
  var diaCts        = props['_diamond_cts'] != null ? parseFloat(props['_diamond_cts']) : null;
  var goldRateOrder = props['_gold_rate']   != null ? parseFloat(props['_gold_rate'])   : null;

  var goldRateLive = null;
  var diaValue     = null;

  // Always fetch variant metafields — needed for C33, B34/C34, and net wt fallback
  if (lineItem.variant_id) {
    var vmData = shopifyGet('variants/' + lineItem.variant_id + '/metafields.json');
    if (vmData && vmData.metafields) {
      var vmf = {};
      vmData.metafields.forEach(function(mf) {
        if (mf.namespace === 'custom') vmf[mf.key] = mf.value;
      });
      if (netWt === null && vmf['net_metal_weight_g']    != null) netWt        = parseFloat(vmf['net_metal_weight_g']);
      if (vmf['gold_rate']              != null)                   goldRateLive = parseFloat(vmf['gold_rate']);
      if (vmf['price_breakup_diamond']  != null)                   diaValue     = parseFloat(vmf['price_breakup_diamond']);
    }
  }

  // Product metafield fallback for diamond carats
  if (diaCts === null && lineItem.product_id) {
    var pmData = shopifyGet('products/' + lineItem.product_id + '/metafields.json');
    if (pmData && pmData.metafields) {
      pmData.metafields.forEach(function(mf) {
        if (mf.namespace === 'custom' && mf.key === 'totaldiamondweight' && diaCts === null) {
          diaCts = parseFloat(mf.value);
        }
      });
    }
  }

  if (netWt         !== null) sheet.getRange(NET_WT_CELL).setValue(netWt);
  if (diaCts        !== null) sheet.getRange(DIA_CTS_CELL).setValue(diaCts);
  if (goldRateOrder !== null) sheet.getRange(GOLD_RATE_ORD_CELL).setValue(goldRateOrder);
  if (diaValue      !== null) {
    sheet.getRange(LGD_RATE_CELL).setValue(diaValue);
    sheet.getRange(DIA_VALUE_CELL).setValue(diaValue);
    // B28 driven by C20 so the 80%/100% rows stay live
    sheet.getRange(DIA_VAL_CELL).setFormula('=' + DIA_VALUE_CELL + '*0.8');
  }
  // GOLD_VAL_CELL is driven by the ONE rate cell so single- and multi-SKU share one formula shape.
  // Single SKU: it is just this item's live rate. Multi: the weighted average.
  // If your gold-value formula includes a karat purity factor (e.g. =B29*(18/24)*C33), adjust here.
  //
  // The rate is written whenever we have one, even with no weight — losing the weight must not also
  // hide what Shopify quoted. The formula is only written when a weight exists to multiply.
  if (goldRateLive !== null) {
    sheet.getRange(GOLD_RATE_EFF_CELL).setValue(goldRateLive);
    sheet.getRange(GOLD_RATE_EFF_CELL).setNote(goldRateNote_(false, null));
    if (netWt !== null) sheet.getRange(GOLD_VAL_CELL).setFormula('=' + NET_WT_CELL + '*' + GOLD_RATE_EFF_CELL);
  }

  var karat = extractKarat(lineItem.sku);
  if (karat) sheet.getRange(KARAT_CELL).setValue(karat);

  // The paid side of the table — what this line item was actually charged at.
  writePaidColumn_(sheet, props, { lineItem: lineItem });

  SpreadsheetApp.flush();
}

// ── READING THE PAID SIDE OFF A LINE ITEM ────────────────────────────────────────
// The price breakup is carried on the order as VISIBLE, Rs-prefixed line-item properties written by
// the middleware's hydrateItemFromVariant / reprice paths:
//
//   Gold             Rs48000.00
//   Diamond          Rs30000.00      (pre-discount — the discount engine reads this as its base)
//   Making           Rs5000.00       ('Making Charges' on older orders)
//   Discount Applied Rs2000.00       pre-tax rupees, positive
//   Gross Value      Rs...           TAX-INCLUSIVE (components × 1.03) — deliberately not used here
//   Taxable Value    Rs...           = Gross Value / 1.03 − Discount Applied
//
// NOT the underscore-prefixed keys. Those are specs (_net_wt, _diamond_cts, _gemstone_cts,
// _gold_rate), not money. The variant metafields price_breakup_gold/_diamond/_making are today's
// catalog price, which is the wrong side of a before/after table — they are the AFTER.
//
// GEMSTONE HAS NO VALUE ANYWHERE. The pricing engine knows only gold, diamond and making; the only
// gemstone fields in the system are weights (_gemstone_cts, jewelcode_gemstone_weight). Its cell is
// left blank on purpose — see the note written onto it by buildCalcTable_.
function paidComponents_(props, lineItem) {
  props = props || {};
  var gold   = rsProp_(props['Gold']);
  var dia    = rsProp_(props['Diamond']);
  var making = rsProp_(props['Making'] !== undefined ? props['Making'] : props['Making Charges']);

  // Fallback for orders predating the hydrate step: reconstruct gold from the rate locked on the
  // line. Only gold can be rebuilt this way — nothing on the order implies diamond or making.
  if (gold === null) {
    var rate = rsProp_(props['_gold_rate']);
    var wt   = rsProp_(props['_net_wt']);
    if (rate !== null && wt !== null) gold = Math.round(rate * wt * 100) / 100;
  }

  // 'Discount Applied' is the middleware's own per-item figure and is preferred over Shopify's
  // allocation, because it is the number the invoice was actually built from.
  var disc = rsProp_(props['Discount Applied']);
  if (disc === null) disc = lineItemDiscount_(lineItem);

  // GST as charged. Without it the paid column sums to the TAXABLE value and sits under a "TOTAL"
  // label that does not match the invoice. Falls back to Shopify's own tax lines for a store that
  // records them; stays null when nothing recorded it, because blank and zero must not look alike.
  var tax = rsProp_(props['GST']);
  if (tax === null) tax = lineItemTax_(lineItem);

  // What was actually paid for the stone — the cap on the Deduction diamond leg. NOT reconstructed
  // as (diamond − discount) when the property is missing: the discount engine takes diamond first
  // and then making, so that subtraction would over-tighten the cap whenever making absorbed part
  // of the cut. Missing property means no cap, which is how the sheet behaved before the cap existed.
  var diaPaid = rsProp_(props['Diamond (After Discount)']);

  return { gold: gold, dia: dia, gem: null, making: making, ship: null, igi: null,
           disc: disc, tax: tax, diaPaid: diaPaid };
}

// Tax Shopify itself recorded against the line. Empty on a tax-inclusive store that writes no tax
// lines (the 'GST' property is the real source there), so null rather than 0 when there is nothing.
function lineItemTax_(lineItem) {
  var total = 0;
  ((lineItem && lineItem.tax_lines) || []).forEach(function (t) { total += toNum(t.price); });
  return total || null;
}

// Parses "Rs48000.00" / "48,000" / 48000 into a number. null when there is nothing to parse, so a
// missing property stays distinguishable from a genuine zero.
function rsProp_(raw) {
  if (raw === null || raw === undefined || raw === '') return null;
  if (typeof raw === 'number') return isNaN(raw) ? null : raw;
  var n = parseFloat(String(raw).replace(/rs\.?/i, '').replace(/,/g, '').trim());
  return isNaN(n) ? null : n;
}

// Total discount allocated to a line item by Shopify. Order-level discounts are apportioned across
// line items in discount_allocations, so this is the item's real share — not the order's whole
// discount, which would over-state a single SKU pulled out of a multi-item order.
function lineItemDiscount_(lineItem) {
  var total = 0;
  ((lineItem && lineItem.discount_allocations) || []).forEach(function (d) { total += toNum(d.amount); });
  if (!total && lineItem && lineItem.total_discount) total = toNum(lineItem.total_discount);
  return total || null;
}



// SKU format: NK00068|Y|18|... — karat is always pipe-index 2
function extractKarat(sku) {
  if (!sku) return '';
  var parts = String(sku).split('|');
  var k     = parts[2] ? parts[2].trim() : '';
  return k ? k + 'K' : '';
}

// DOCTYPE_CELL (B37) = "Voucher" (1-year store credit, discount code) or
// "Exchange Note" (instant post-tax deduction on a new invoice).
function createDocument() {
  const calc = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CALC_SHEET_NAME);
  // Source wins over Document Type: Old Gold is always a voucher bound to a customer, no order.
  if (currentSource_(calc) === SOURCE_OLD_GOLD) return createOldGoldVoucher_();
  const modality = String(calc.getRange(DOCTYPE_CELL).getValue()).trim().toLowerCase();
  if (modality.indexOf('exchange') === 0 || modality === 'exc') return createExchangeNote_();
  return createVoucher_();
}

// ── VOUCHER (rebranded credit note) — 1-year discount code, tagged to the order ──
function createVoucher_() {
  const ss   = SpreadsheetApp.getActiveSpreadsheet();
  const calc = ss.getSheetByName(CALC_SHEET_NAME);
  const log  = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
  const ui   = SpreadsheetApp.getUi();
  if (!log) { ui.alert('Log tab "' + VOUCHER_LOG + '" not found. Run "Set up Document Type fields" first.'); return; }

  const customerName  = calc.getRange(CUST_NAME_CELL).getValue();
  const customerEmail = calc.getRange(CUST_EMAIL_CELL).getValue();
  const orderNumber   = String(calc.getRange(ORDER_NUM_CELL).getValue()).trim();
  const netWt         = toNum(calc.getRange(NET_WT_CELL).getValue());
  const diaWt         = toNum(calc.getRange(DIA_CTS_CELL).getValue());
  const goldVal       = toNum(calc.getRange(GOLD_VAL_CELL).getValue());
  const diaVal        = toNum(calc.getRange(DIA_VAL_CELL).getValue());
  refreshFinalIssueValue_(calc);
  const netCredit     = toNum(calc.getRange(FINAL_ISSUE_CELL).getValue());
  if (!fullValueGuardPasses_(calc, ui, orderNumber, netCredit)) return;

  const today      = new Date();
  // 1-year validity (same day next year — JS rolls leap years over cleanly).
  const validUntil = new Date(today.getFullYear() + 1, today.getMonth(), today.getDate());

  if (!customerEmail || !orderNumber || netCredit <= 0) {
    ui.alert('Missing data. Fill customer email, order number, and ensure net credit > 0.');
    return;
  }

  const year   = today.getFullYear();
  // Serial from the central counter service (atomic, no gaps/dupes across devices).
  // Falls back to the legacy sheet-row count if the middleware is unreachable.
  // Print the ledger's own code (VCH27-KAHSR-0001) — never re-format locally. Middleware down →
  // legacy sheet-row format so a voucher can still be issued; reconcile that number by hand after.
  const alloc  = allocateVoucherSerial();
  const cnNum  = (alloc && alloc.serial_code)
                 ? alloc.serial_code
                 : 'VCH-' + year + '-' + String(log.getLastRow()).padStart(4, '0');

  const issued    = Utilities.formatDate(today, 'Asia/Kolkata', 'dd-MM-yyyy');
  const expiryFmt = Utilities.formatDate(validUntil, 'Asia/Kolkata', 'dd-MM-yyyy');
  const expiryIso = validUntil.toISOString();

  const cleanOrderNum = orderNumber.replace('#', '');
  const orderData     = getOrderData(cleanOrderNum);
  const orderId       = orderData ? orderData.id        : null;
  const customerId    = orderData ? orderData.customerId : null;

  const priceRulePayload = {
    title:              cnNum,
    target_type:        'line_item',
    target_selection:   'all',
    allocation_method:  'across',
    value_type:         'fixed_amount',
    value:              '-' + netCredit.toFixed(2),
    customer_selection: customerId ? 'prerequisite' : 'all',
    starts_at:          today.toISOString(),
    ends_at:            expiryIso,
    usage_limit:        1
  };
  if (customerId) priceRulePayload.prerequisite_customer_ids = [customerId];

  const priceRule = shopifyPost('price_rules.json', { price_rule: priceRulePayload });
  if (!priceRule || !priceRule.price_rule) {
    ui.alert('Failed to create price rule in Shopify. Check Supabase credentials and token scopes.');
    return;
  }

  const priceRuleId = priceRule.price_rule.id;
  const discCode    = shopifyPost('price_rules/' + priceRuleId + '/discount_codes.json', {
    discount_code: { code: cnNum }
  });
  if (!discCode || !discCode.discount_code) {
    ui.alert('Price rule created but discount code failed. Check Shopify manually.');
    return;
  }

  calc.getRange(DOCNUM_OUT_CELL).setValue(cnNum);

  // Record the voucher in the credit-instrument ledger. WITHOUT this the voucher exists only as a
  // Shopify discount code: handleApplyVoucherTag looks it up via getBySerial and, finding nothing,
  // tags the draft "voucher-invalid: ... not found" and deducts nothing. The Exchange Note gets its
  // ledger row for free inside /api/exc-redeem; the voucher has no such call, so it must issue here.
  if (!issueCreditInstrument_({
        instrumentType:  'voucher',
        serialCode:      cnNum,
        value:           netCredit,
        customerId:      customerId ? String(customerId) : null,
        customerName:    customerName,
        sourceOrderName: orderNumber,
        stateCode:       resolveStoreCode_(),
        expiresAt:       expiryIso,
        priceRuleId:     String(priceRuleId)
      })) {
    ui.alert('⚠️ ' + cnNum + ' was created in Shopify but NOT recorded in the ledger.\n\n' +
             'It will fail to apply from the admin app ("not found"). Re-run once the middleware is reachable.');
  }

  // Optionally spend the voucher on the sale being rung up right now — the voucher twin of what
  // createExchangeNote_ has always done. NEWDRAFT_CELL used to be read ONLY by the exchange-note
  // path, so a staff member who typed a draft number and left Document Type on "Voucher" got a
  // voucher that silently deducted nothing from that draft and no error anywhere. Blank is still
  // the normal case: a voucher is valid standalone store credit, spent later from the admin app.
  const applyRes = applyVoucherIfDraftGiven_(calc, cnNum, netCredit, orderNumber, customerName);

  // Internal cn-* tag names kept unchanged so the existing OPP print template renders untouched.
  if (orderId) {
    addOrderTags(orderId, [
      'cn-issued',
      'cn-num:' + cnNum,
      'cn-val:' + netCredit.toFixed(2),
      'cn-exp:' + expiryFmt,
      'cn-iss:' + issued
    ]);
  } else {
    ui.alert('⚠️ Order ' + orderNumber + ' not found in Shopify. Voucher created but order not tagged.');
  }

  // Last column (M) holds the Shopify price_rule_id so a later Void can delete the discount.
  log.appendRow([issued, cnNum, orderNumber, customerName, customerEmail,
                 netWt, diaWt, goldVal, diaVal, netCredit, expiryFmt, applyRes.status, String(priceRuleId)]);

  var ctx = afterSalesContext_(calc);
  logAfterSales_({
    'Logged At': issued, 'Event': 'issued',
    'Source': ctx.Source, 'Exchange Type': ctx['Exchange Type'], 'Doc Type': ctx['Doc Type'], 'Store': ctx.Store,
    'Serial': cnNum, 'Order': orderNumber,
    'Applied To': String(calc.getRange(NEWDRAFT_CELL).getValue()).trim(),
    'Value': netCredit, 'Net Wt': netWt, 'Dia Cts': diaWt, 'Gold Value': goldVal, 'Diamond Value': diaVal,
    'Customer': customerName, 'Email': customerEmail, 'Expiry': expiryFmt,
    'Status': applyRes.error ? 'issued (apply failed)' : 'outstanding',
    'Reason / Note': String(calc.getRange(OVERRIDE_REASON_CELL).getValue()).trim() || applyRes.error,
    'Ref': String(priceRuleId)
  });

  sendVoucherEmail_(customerName, customerEmail, cnNum, netCredit, expiryFmt, orderNumber);

  ui.alert(
    (applyRes.error ? '⚠️ Voucher Created — but NOT applied\n\n' : '✅ Voucher Created\n\n') +
    'Voucher: ' + cnNum + '\n' +
    'Discount Code: ' + cnNum + '\n' +
    'Value: ₹' + netCredit.toLocaleString('en-IN', { minimumFractionDigits: 2 }) + '\n' +
    'Valid Until: ' + expiryFmt + ' (1 year)\n\n' +
    applyRes.message +
    'Order ' + orderNumber + ' tagged. ' +
    (voucherMailOn_() ? 'Email sent to ' + customerEmail + '.' : '✉️ Email SUPPRESSED (test mode).')
  );
}

// ── OLD-GOLD VOUCHER — buy-back credit, no reference order, no draft ──────────────
// Customer walks in with scrap/old gold. Staff enter weight + purity; it's valued off the same
// buying_rate_table the middleware uses, and issued as a 1-year voucher BOUND TO THE CUSTOMER
// (prerequisite_customer_ids). No order, no draft — the voucher stands alone and is redeemed later
// against a future sale. The customer is resolved from OG_CUSTID_CELL (populate it via the
// "Look up Old-Gold customer" menu item, which calls lookupCustomer_).
function createOldGoldVoucher_() {
  const ss   = SpreadsheetApp.getActiveSpreadsheet();
  const calc = ss.getSheetByName(CALC_SHEET_NAME);
  const log  = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
  const ui   = SpreadsheetApp.getUi();
  if (!log) { ui.alert('Log tab "' + VOUCHER_LOG + '" not found. Run the setup first.'); return; }

  // Customer: "id | name" written into OG_CUSTID_CELL by the lookup. Require a resolved id — an
  // old-gold voucher with no customer would be a bearer instrument redeemable by anyone.
  const custRaw = String(calc.getRange(OG_CUSTID_CELL).getValue()).trim();
  const custId  = custRaw.split('|')[0].trim();
  const custNm  = (custRaw.split('|')[1] || '').trim() || String(calc.getRange(CUST_NAME_CELL).getValue()).trim();
  if (!custId || !/^\d+$/.test(custId)) {
    ui.alert('No customer resolved.\n\nEnter a phone or email in ' + OG_CUSTOMER_CELL +
             ' and run "🔎 Look up Old-Gold customer" from the menu first.');
    return;
  }

  const weight = toNum(calc.getRange(OG_WEIGHT_CELL).getValue());
  const purity = toNum(calc.getRange(OG_PURITY_CELL).getValue());
  const rate   = toNum(calc.getRange(OG_RATE_CELL).getValue()) || getBuyingRate_(purity);
  if (!(weight > 0) || !(purity > 0)) { ui.alert('Enter old-gold weight (' + OG_WEIGHT_CELL + ') and purity (' + OG_PURITY_CELL + ').'); return; }
  if (!(rate > 0)) { ui.alert('No buy-back rate for ' + purity + 'kt. Check the buying rate table is set.'); return; }

  // Old-gold value = weight × buy-back rate, computed here at issue time. It does NOT read or write
  // the NET cell (that cell is the Purchase-Exchange formula; touching it would break the exchange
  // calc). Any note typed in OVERRIDE_REASON_CELL is recorded on the log for audit.
  const value  = Math.round(weight * rate * 100) / 100;
  const reason = String(calc.getRange(OVERRIDE_REASON_CELL).getValue()).trim();

  const today      = new Date();
  const validUntil = new Date(today.getFullYear() + 1, today.getMonth(), today.getDate());
  const issued     = Utilities.formatDate(today, 'Asia/Kolkata', 'dd-MM-yyyy');
  const expiryFmt  = Utilities.formatDate(validUntil, 'Asia/Kolkata', 'dd-MM-yyyy');
  const expiryIso  = validUntil.toISOString();
  const year       = today.getFullYear();

  const alloc = allocateVoucherSerial();
  const cnNum = (alloc && alloc.serial_code)
                ? alloc.serial_code
                : 'VCH-' + year + '-' + String(log.getLastRow()).padStart(4, '0');

  // Customer-bound, single-use price rule — identical shape to createVoucher_, minus the order link.
  const priceRule = shopifyPost('price_rules.json', { price_rule: {
    title:              cnNum,
    target_type:        'line_item',
    target_selection:   'all',
    allocation_method:  'across',
    value_type:         'fixed_amount',
    value:              '-' + value.toFixed(2),
    customer_selection: 'prerequisite',
    prerequisite_customer_ids: [Number(custId)],
    starts_at:          today.toISOString(),
    ends_at:            expiryIso,
    usage_limit:        1
  }});
  if (!priceRule || !priceRule.price_rule) { ui.alert('Failed to create the price rule in Shopify. Check token scopes.'); return; }
  const priceRuleId = priceRule.price_rule.id;
  const discCode = shopifyPost('price_rules/' + priceRuleId + '/discount_codes.json', { discount_code: { code: cnNum } });
  if (!discCode || !discCode.discount_code) { ui.alert('Price rule created but discount code failed. Check Shopify.'); return; }

  calc.getRange(DOCNUM_OUT_CELL).setValue(cnNum);

  if (!issueCreditInstrument_({
        instrumentType:  'voucher',
        serialCode:      cnNum,
        value:           value,
        customerId:      String(custId),
        customerName:    custNm,
        sourceOrderName: 'OLD-GOLD ' + weight.toFixed(3) + 'g @ ' + purity + 'kt',
        stateCode:       resolveStoreCode_(),
        expiresAt:       expiryIso,
        priceRuleId:     String(priceRuleId)
      })) {
    ui.alert('⚠️ ' + cnNum + ' created in Shopify but NOT recorded in the ledger. It will fail to apply. Re-run when the middleware is reachable.');
  }

  // Same optional "spend it on this sale now" step as createVoucher_. Old gold has no originating
  // order, so nothing is passed as the linkage reference — the draft gets vch-applied / vch-num only.
  const applyRes = applyVoucherIfDraftGiven_(calc, cnNum, value, '', custNm);

  // Log columns match the Voucher Log header; old-gold specifics go in the weight/gold-value slots.
  log.appendRow([issued, cnNum, 'OLD-GOLD', custNm, calc.getRange(CUST_EMAIL_CELL).getValue(),
                 weight, 0, value, 0, value, expiryFmt, applyRes.status, String(priceRuleId), reason]);

  logAfterSales_({
    'Logged At': issued, 'Event': 'issued',
    'Source': 'Old Gold', 'Exchange Type': '—', 'Doc Type': 'Voucher', 'Store': resolveStoreCode_(),
    'Serial': cnNum, 'Order': 'OLD-GOLD',
    'Applied To': String(calc.getRange(NEWDRAFT_CELL).getValue()).trim(),
    'Value': value, 'Net Wt': weight, 'Gold Value': value,
    'Customer': custNm, 'Email': calc.getRange(CUST_EMAIL_CELL).getValue(), 'Expiry': expiryFmt,
    'Status': applyRes.error ? 'issued (apply failed)' : 'outstanding',
    // Old gold's "why" is the purity/rate it was bought back at — the report cannot re-derive it,
    // because the buy-back table moves daily.
    'Reason / Note': weight.toFixed(3) + 'g @ ' + purity + 'kt × ₹' + rate.toFixed(2) + '/g' +
                     (reason ? ' — ' + reason : ''),
    'Ref': String(priceRuleId)
  });

  sendVoucherEmail_(custNm, calc.getRange(CUST_EMAIL_CELL).getValue(), cnNum, value, expiryFmt, 'OLD-GOLD');

  ui.alert(
    (applyRes.error ? '⚠️ Old-Gold Voucher Created — but NOT applied\n\n' : '✅ Old-Gold Voucher Created\n\n') +
    'Voucher: ' + cnNum + '\n' +
    'Customer: ' + custNm + ' (id ' + custId + ')\n' +
    'Old gold: ' + weight.toFixed(3) + ' g @ ' + purity + 'kt × ₹' + rate.toFixed(2) + '/g\n' +
    'Value: ₹' + value.toLocaleString('en-IN', { minimumFractionDigits: 2 }) +
    (reason ? '  (override: ' + reason + ')' : '') + '\n' +
    'Valid Until: ' + expiryFmt + ' (1 year)\n\n' +
    applyRes.message +
    (voucherMailOn_() ? 'Email sent.' : '✉️ Email SUPPRESSED (test mode).')
  );
}

// Menu action: value the old gold from purity in OG_PURITY_CELL → rate into OG_RATE_CELL, and show
// the resulting value. It is deliberately NOT written to the NET cell (NET_VALUE_CELL is a live
// formula for Purchase Exchange — writing a static number there would destroy that formula). The
// old-gold value is computed fresh at issue time in createOldGoldVoucher_; here it's just surfaced.
function fetchOldGoldRate() {
  var calc = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  var purity = toNum(calc.getRange(OG_PURITY_CELL).getValue());
  if (!(purity > 0)) { ui.alert('Enter the old-gold purity (karat) in ' + OG_PURITY_CELL + ' first.'); return; }
  var rate = getBuyingRate_(purity);
  if (!(rate > 0)) { ui.alert('No buy-back rate for ' + purity + 'kt — check the buying rate table is set in Supabase.'); return; }
  calc.getRange(OG_RATE_CELL).setValue(rate);
  var value = oldGoldValue_(calc);
  showOldGoldValue_(calc, value);
  var weight = toNum(calc.getRange(OG_WEIGHT_CELL).getValue());
  ui.alert('Buy-back rate for ' + purity + 'kt: ₹' + rate.toFixed(2) + '/g' +
           (weight > 0 ? '\nOld-gold value for ' + weight.toFixed(3) + ' g: ₹' + value.toFixed(2) : ''));
}

// weight × rate, rounded to paise. Pure — reads the OG cells, writes nothing.
function oldGoldValue_(calc) {
  var weight = toNum(calc.getRange(OG_WEIGHT_CELL).getValue());
  var rate   = toNum(calc.getRange(OG_RATE_CELL).getValue());
  return Math.round(weight * rate * 100) / 100;
}

// Surface the computed old-gold value WITHOUT touching the NET formula: a note on the rate cell.
function showOldGoldValue_(calc, value) {
  if (value > 0) calc.getRange(OG_RATE_CELL).setNote('Old-gold value = weight × rate = ₹' + value.toFixed(2));
}

// One-time setup: builds the Source / Exchange-Type / Old-Gold field block below B47, and LOCKS the
// calculated cells (weights, carats, rates, computed values) so staff can only edit the final value
// (B36) — with a reason in OVERRIDE_REASON_CELL. Idempotent; refuses to overwrite non-ours content.
// ── MIGRATION: move all controls to a block at the TOP of the sheet ──────────────
// Legacy layout ran Customer → Item → Weights → Rates → Deductions → Calc, with the controls
// (Doc Type, Source, Old-Gold, Store) bolted on at the bottom. This inserts a CONTROLS block at
// the very top by inserting RESTRUCTURE_ROWS rows at row 1 — which shifts every legacy row down by
// that amount and, crucially, lets Google Sheets AUTO-ADJUST every calculation formula (the
// 80%/100%/deduction rows) so the numbers are preserved exactly. It then builds the control fields
// in the new top rows and clears the now-shifted legacy control cells. The layout constants at the
// top of this file already point at the post-migration positions, so run this ONCE.
//
// TEST ON A COPY FIRST (File → Make a copy). Idempotent-guarded: re-running is a no-op.

// ── FRESH BUILD: construct the whole Exchange Calculator from scratch, controls-on-top ────────────
// Deterministic reset. Rebuilds every label, dropdown, and CALC FORMULA at the exact cells the layout
// constants point to — so the sheet always matches the script and nothing can be left corrupted. Run
// this instead of the migration if the sheet's formulas got damaged. Input cells are left blank; the
// populate/auto-fill logic fills them from an order as normal.
//
// Calc convention (matches the original): Gold at 100%, Diamond at 80%, minus deductions.
//   Gross Exchange Value = Gold Value + Diamond Value × 80%
//   NET                  = Gross − Total Deductions
function buildExchangeCalculator() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  var confirm = ui.alert('Rebuild the Exchange Calculator?',
    'This rewrites all labels, dropdowns and calc formulas on the "' + CALC_SHEET_NAME + '" tab to the ' +
    'controls-on-top layout. Input data on the tab is cleared. The log tabs are untouched.\n\nProceed?',
    ui.ButtonSet.YES_NO);
  if (confirm !== ui.Button.YES) return;

  // Clear the working area (labels A, values B, helper cols C/D) down to the details block.
  calc.getRange('A1:D60').clearContent().clearDataValidations().clearNote();
  calc.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(function (p) {
    try { if (/Timanti CN/.test(p.getDescription())) p.remove(); } catch (e) {}
  });

  function label(a1, text) { var r = calc.getRange(a1); calc.getRange(r.getRow(), 1).setValue(text); }
  function header(row, text) { calc.getRange(row, 1).setValue(text); }
  function drop(a1, list, def) {
    var r = calc.getRange(a1);
    r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(list, true).setAllowInvalid(false).build());
    if (def) r.setValue(def);
  }

  // ── CONTROLS (top) ──
  header(1, '0 — CONTROLS  (set these first)');
  label(SOURCE_CELL,   'Source');            drop(SOURCE_CELL,   ['Purchase Exchange', 'Old Gold'], 'Purchase Exchange');
  label(EXCHTYPE_CELL, 'Exchange Type');     drop(EXCHTYPE_CELL, ['Deduction', 'Full Value'],       'Deduction');
  label(DOCTYPE_CELL,  'Document Type');     drop(DOCTYPE_CELL,  ['Voucher', 'Exchange Note'],       'Voucher');
  label(STORE_CODE_CELL, 'Store Code');      drop(STORE_CODE_CELL, ['KA-HSR', 'KA-TEST'], 'KA-HSR');
  refreshNewDraftField_(calc);               // label + note + clears any legacy gray-out rule
  header(cellRow_(OG_CUSTOMER_CELL) - 1, '— OLD GOLD  (only when Source = Old Gold) —');
  label(OG_CUSTOMER_CELL, 'Old Gold — Customer phone/email');
  label(OG_CUSTID_CELL,   'Old Gold — Customer (auto)');
  label(OG_WEIGHT_CELL,   'Old Gold — Weight (g)');
  label(OG_PURITY_CELL,   'Old Gold — Purity (karat)');
  label(OG_RATE_CELL,     'Old Gold — Buy-back rate/g (auto)');
  label(OVERRIDE_REASON_CELL, 'Final value override — reason');

  // ── 1: Customer & order ──
  header(cellRow_(CUST_NAME_CELL) - 1, '1 — CUSTOMER & ORDER DETAILS');
  label(CUST_NAME_CELL, 'Customer Name');
  label(CUST_EMAIL_CELL, 'Customer Email');
  label(CUST_PHONE_CELL, 'Customer Phone');
  label(ORDER_NUM_CELL, 'Original Order Number (e.g. #1020)');
  label(ORDER_DATE_CELL, 'Original Order Date');
  header(cellRow_(SKU_CELL) - 1, 'Item Being Exchanged');
  label(SKU_CELL, 'SKU / Jewellery Code');
  label(KARAT_CELL, 'Metal Karat');

  // ── 2: Weights ──
  header(cellRow_(NET_WT_CELL) - 1, '2 — WEIGHTS');
  label(NET_WT_CELL, 'Net Gold Weight (g)');
  label(DIA_CTS_CELL, 'Total Diamond Weight (cts)');

  // ── 3: Rates ──
  header(cellRow_(GOLD_RATE_ORD_CELL) - 1, '3 — TODAY\'S RATES');
  label(GOLD_RATE_ORD_CELL, GOLD_RATE_LABEL);
  label(LGD_RATE_CELL, 'Diamond Value (from variant →)');
  label(DIA_PAID_CELL, 'Diamond paid (after discount)');

  // ── 4 + 5: Calculation (before / after) ──
  header(cellRow_(GOLD_VAL_CELL) - 1, '4 — CALCULATION  (auto)');

}

// ── THE BEFORE/AFTER CALCULATION TABLE ───────────────────────────────────────────
// Extracted from the fresh build so the in-place migration lays out byte-identical cells. Writes
// labels, the column headings, the two totals and every formula — but NEVER a component figure.
// Those arrive when an order is populated, so re-running this on a live sheet re-labels without
// wiping the numbers a half-finished exchange is holding.
function buildCalcTable_(calc) {
  var first = cellRow_(CALC_ROWS[0].value);            // 41
  var last  = cellRow_(CALC_ROWS[CALC_ROWS.length - 1].value);  // 48
  var total = cellRow_(PAID_TOTAL_CELL);               // 49

  // Column headings sit on the section header row, directly above the first component.
  calc.getRange(first - 1, 2).setValue('Paid at order');
  calc.getRange(first - 1, 3).setValue('Exchange value');
  calc.getRange(first - 1, WHY_COL).setValue('Why');
  calc.getRange(first - 1, 2, 1, 3).setFontWeight('bold');

  CALC_ROWS.forEach(function (r) {
    var row = cellRow_(r.value);
    calc.getRange(row, 1).setValue(r.label);
    calc.getRange(row, WHY_COL).setValue(r.whyDed).setFontColor('#777777').setFontStyle('italic');
  });

  // Column C — every row of it, including the discount cell whose correct state is "empty". That
  // matters: an earlier revision added the discount back with =-B47, and a formula lives in the
  // cell, not in the code, so re-pasting the script does not remove it and neither does relabelling
  // the row. The builder has to be authoritative about the whole column or a rebuild silently
  // leaves old behaviour running under a label that now says the opposite.
  refreshExchangeColumn_(calc);
  // Say why the gemstone paid cell is empty, on the cell itself. Nothing in the catalogue or the
  // pricing engine carries a gemstone VALUE — only weights — so there is no figure to read. Left
  // unexplained this looks like the lookup failing rather than the data not existing.
  calc.getRange(PAID_GEM_CELL).setNote(
    'Blank because no gemstone VALUE exists anywhere in the system.\n\n' +
    'The pricing engine breaks price into gold, diamond and making only. The gemstone fields ' +
    '(_gemstone_cts, jewelcode_gemstone_weight) are WEIGHTS, not rupees.\n\n' +
    'If gemstone is priced separately, it needs a price_breakup_gemstone variant metafield and a ' +
    '"Gemstone" line-item property before this cell can fill itself.');

  // Totals. The paid total is informational; the exchange total is the number that matters.
  calc.getRange(total, 1).setValue('TOTAL');
  calc.getRange(total, 2).setFormula('=SUM(B' + first + ':B' + last + ')');
  calc.getRange(total, 3).setFormula('=SUM(C' + first + ':C' + last + ')');
  calc.getRange(total, 1, 1, 3).setFontWeight('bold');
  calc.getRange(total, WHY_COL).setValue('what the item was worth then vs now')
      .setFontColor('#777777').setFontStyle('italic');

  // NET stays a distinct, freely-editable row — it is the one place staff may overrule the
  // calculation, and collapsing it into the total would remove that escape hatch.
  calc.getRange(cellRow_(NET_VALUE_CELL), 1).setValue('NET (Deduction calc)');
  calc.getRange(NET_VALUE_CELL).setFormula('=C' + total);

  // The value actually issued — mirrors NET for Deduction, the original-invoice taxable for
  // Full Value, and weight × rate for Old Gold. Maintained live by refreshFinalIssueValue_.
  calc.getRange(cellRow_(FINAL_ISSUE_CELL), 1).setValue('➡ VALUE TO ISSUE (auto)');
  calc.getRange(FINAL_ISSUE_CELL).setFormula('=' + NET_VALUE_CELL);
}

// Writes the PAID column and the not-credited zeros from one order line item. Called by both
// populate paths. Everything here is read from line-item PROPERTIES — the price frozen at purchase —
// never from the variant, because the variant tells you today's price, not what was charged.
//
// A missing property leaves its cell BLANK rather than writing 0. Blank reads as "the order did not
// record this"; 0 reads as "the customer was charged nothing", and those must not look alike on a
// sheet someone signs a credit note from.
function writePaidColumn_(sheet, props, opts) {
  opts = opts || {};
  var paid = opts.totals || paidComponents_(props, opts.lineItem);
  var put  = function (cell, v) {
    if (v === null || v === undefined || isNaN(v)) sheet.getRange(cell).clearContent();
    else sheet.getRange(cell).setValue(v);
  };

  put(PAID_GOLD_CELL,   paid.gold);
  put(PAID_DIA_CELL,    paid.dia);
  put(PAID_GEM_CELL,    paid.gem);
  put(PAID_MAKING_CELL, paid.making);
  put(PAID_SHIP_CELL,   paid.ship);
  put(PAID_IGI_CELL,    paid.igi);
  // Discount is held NEGATIVE — it reduced the invoice. Every source reports it positive, so it is
  // flipped once here rather than at each call site.
  put(PAID_DISC_CELL, paid.disc === null || paid.disc === undefined ? null : -Math.abs(paid.disc));
  put(PAID_TAX_CELL,  paid.tax);
  put(DIA_PAID_CELL,  paid.diaPaid);

  // Column C is written in exactly one place, for whichever mode is selected.
  refreshExchangeColumn_(sheet);
}

// ── THE ONE WRITER FOR COLUMN C ──────────────────────────────────────────────────
// Writes the whole exchange column for the mode selected in SOURCE_CELL / EXCHTYPE_CELL. Every path
// that can change the answer — populating an order, flipping either dropdown, issuing a document —
// comes through here, so Deduction and Full Value cannot drift apart.
//
// DEDUCTION  — gold at 100% of today's rate, diamond at 80% of today's catalogue value capped at
//              what was paid for the stone, and a hard zero on every other row. The discount row is
//              left BLANK: see DISCOUNT_ROW_NOTE for why it is not deducted a second time.
// FULL VALUE — every row mirrors its paid cell, so the column totals exactly what the customer was
//              charged, tax inclusive.
//
// Custom deductions is never touched in either mode — it is the one cell in this column staff own.
//
// Blank is preserved as blank on the Full Value side. A plain '=B43' renders 0, and a zero in the
// gemstone row would claim the customer was charged nothing for a gemstone when the truth is that
// the order never recorded one. Those must not look alike on a sheet someone signs a credit note from.
function refreshExchangeColumn_(calc) {
  var full = isFullValue_(calc);

  CALC_ROWS.forEach(function (r) {
    var cell = calc.getRange(r.value);
    calc.getRange(cellRow_(r.value), WHY_COL)
        .setValue(full ? r.whyFull : r.whyDed).setFontColor('#777777').setFontStyle('italic');

    if (r.key === 'custom') return;                       // staff's own cell, in both modes
    if (full) { cell.setFormula('=IF(' + r.paid + '="","",' + r.paid + ')'); return; }

    switch (r.key) {
      case 'gold':
        cell.setFormula('=' + NET_WT_CELL + '*' + GOLD_RATE_EFF_CELL);
        break;
      case 'dia':
        cell.setFormula('=IF(' + DIA_PAID_CELL + '="",' + DIA_VALUE_CELL + '*0.8,' +
                        'MIN(' + DIA_VALUE_CELL + '*0.8,' + DIA_PAID_CELL + '))');
        break;
      case 'disc':
        cell.clearContent();                              // absorbed by the haircut
        break;
      default:
        cell.setValue(0);                                 // gemstone / making / shipping / IGI / GST
    }
  });

  calc.getRange(DIA_VAL_CELL).setNote(DIA_CAP_NOTE);
  calc.getRange(DISCOUNT_VAL_CELL).setNote(DISCOUNT_ROW_NOTE);
  calc.getRange(TAX_VAL_CELL).setNote(TAX_ROW_NOTE);
}

// True when the Exchange Type dropdown is on Full Value.
//
// Matches the word ANYWHERE in the cell, not only at the start. The test used to be
// indexOf('full') === 0, so any wording that did not LEAD with "Full" — a sheet whose dropdown had
// been relabelled "Purchase Exchange Full Value", for instance — fell silently into the Deduction
// branch and issued the deduction figure under a Full Value label, with no warning anywhere.
function isFullValue_(calc) {
  if (currentSource_(calc) === SOURCE_OLD_GOLD) return false;   // no invoice to take at full value
  return String(calc.getRange(EXCHTYPE_CELL).getValue())
           .trim().toLowerCase().indexOf(EXCHTYPE_FULL) !== -1;
}

// ── MIGRATION: add the GST row and the diamond-paid cap cell ──────────────────────
// The calc block gains one row. GST belongs between the discount and custom deductions, because
// that is where it sits on an invoice — tax is charged on the post-discount taxable — so the row is
// INSERTED rather than appended, and everything below it moves down one:
//
//   Custom deductions  48 → 49      NET            50 → 51
//   TOTAL              49 → 50      VALUE TO ISSUE 51 → 52
//
// Sheets rewrites every formula that referenced the moved rows, and buildCalcTable_ then rewrites
// the totals from the constants, so the arithmetic survives the move on its own.
//
// The cap cell needs no insert: rows 35-39 under "TODAY'S RATES" were already empty.
// Idempotent — safe to run twice.
function migrateAddTaxRow() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  var taxRow   = cellRow_(PAID_TAX_CELL);      // 48 — where GST goes
  var totalRow = cellRow_(PAID_TOTAL_CELL);    // 50 — where TOTAL ends up

  var labelAt = function (row) { return String(calc.getRange(row, 1).getValue()).trim(); };

  if (labelAt(taxRow) === 'GST' && labelAt(totalRow) === 'TOTAL') {
    buildCalcTable_(calc);                     // re-label / re-note only; no row moves
    lockCalcCells_(calc);
    SpreadsheetApp.flush();
    ui.alert('Already on the current layout.\n\nLabels, notes and formulas were refreshed. ' +
             'Nothing moved.');
    return;
  }

  // A sheet ready for this migration has TOTAL one row above where it is going to end up.
  if (labelAt(totalRow - 1) !== 'TOTAL') {
    ui.alert('Aborted — this does not look like the expected layout.\n\n' +
             'Row ' + (totalRow - 1) + ' should read "TOTAL" and reads "' + labelAt(totalRow - 1) + '".\n\n' +
             'Nothing was changed. Run "Rebuild calculator" on a fresh sheet, or tell me what row ' +
             'the TOTAL is on.');
    return;
  }

  // Read the pre-migration NET so the alert can prove the value did not move. It is still one row
  // above its destination at this point.
  var beforeNet = calc.getRange(cellRow_(NET_VALUE_CELL) - 1, 3).getValue();

  calc.insertRowBefore(taxRow);
  buildCalcTable_(calc);
  calc.getRange(cellRow_(DIA_PAID_CELL), 1).setValue('Diamond paid (after discount)');
  lockCalcCells_(calc);
  SpreadsheetApp.flush();

  ui.alert('✅ GST row added.\n\n' +
           'GST            → ' + PAID_TAX_CELL + ' / ' + TAX_VAL_CELL + '  (new)\n' +
           'Diamond paid   → ' + DIA_PAID_CELL + '  (new, caps the diamond leg)\n' +
           'TOTAL moved    → row ' + totalRow + '\n' +
           'NET moved      → ' + NET_VALUE_CELL + '\n' +
           'VALUE TO ISSUE → ' + FINAL_ISSUE_CELL + '\n\n' +
           'NET before this ran: ' + beforeNet + '\n\n' +
           'Re-enter the order number in ' + ORDER_NUM_CELL + ' to fill the new cells, then check ' +
           'that the paid column now totals what the customer was actually charged.');
}

// ── MIGRATION: turn the old single-column calc into the before/after table ────────
// The old block ran Gold, Diamond, Gross, then five "Less:" rows down column B. The new one puts the
// paid figures in B and the live calculation in C, and adds a Gemstone row the old layout had no
// slot for. Idempotent.
//
// Column B is NOT cleared. On an old sheet the "Less:" rows are blank anyway (confirmed with the
// founder: B44 was never filled on any past voucher or exchange note), and anything that IS there is
// a staff figure worth keeping in front of them rather than deleting on their behalf.
function migrateCalcToBeforeAfter() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  // This targets the ORIGINAL single-column layout only. On a sheet that already has the
  // before/after table it would lay a 9-row block over an 8-row one and shift every figure by a
  // row, so it refuses rather than corrupting a live sheet.
  if (String(calc.getRange(cellRow_(PAID_TAX_CELL), 1).getValue()).trim() === 'GST') {
    ui.alert('Already on the before/after layout — nothing to do.\n\n' +
             'Use "Add GST row + diamond cap" if you are looking for the current migration.');
    return;
  }

  // Read the pre-migration figures BEFORE anything is rewritten, so the alert can prove the value
  // did not move. The old NET lived at B50, the old gross at B43.
  var beforeNet = calc.getRange('B50').getValue();

  // Strip every leftover FORMULA from the paid column FIRST — before buildCalcTable_ writes the new
  // paid total into B49, which this would otherwise wipe straight back out.
  //
  // The old layout computed down column B: gold at B41, diamond at B42, gross at B43, total
  // deductions at B49, NET at B50, issue at B51. Column B is now data, not arithmetic.
  //
  // Not cosmetic. B41 still reads =B29*D33, and migrateGoldRateCell clears D33 — leave it and the
  // "Paid at order" gold cell silently becomes ₹0, a wrong number in the one column staff are meant
  // to trust as history. Same for B42 once the diamond source moves.
  //
  // Only formulas go. Anything a staff member TYPED into B44-B48 is a constant and survives —
  // deleting someone's working figures on their behalf is not this function's business.
  CALC_ROWS.map(function (r) { return r.paid; }).concat([PAID_TOTAL_CELL])
    .forEach(function (a1) {
      if (String(calc.getRange(a1).getFormula())) calc.getRange(a1).clearContent();
    });

  // B50/B51 are cleared UNCONDITIONALLY, formula or not. They sit below the paid table and hold
  // nothing in the new layout, and the old VALUE TO ISSUE was a STATIC number whenever the sheet was
  // last used in Full Value or Old Gold mode — refreshFinalIssueValue_ writes a value there, not a
  // formula. A formula-only sweep would leave that number stranded in the paid column, where it
  // reads as "what the customer paid" and is nothing of the sort.
  ['B' + cellRow_(NET_VALUE_CELL), 'B' + cellRow_(FINAL_ISSUE_CELL)].forEach(function (a1) {
    calc.getRange(a1).clearContent();
    calc.getRange(a1).clearNote();
  });

  buildCalcTable_(calc);

  lockCalcCells_(calc);
  SpreadsheetApp.flush();

  ui.alert('✅ Calculation rebuilt as a before/after table.\n\n' +
           'B = paid at order · C = exchange value · D = why\n\n' +
           'NET moved B50 → ' + NET_VALUE_CELL + '\n' +
           'VALUE TO ISSUE moved B51 → ' + FINAL_ISSUE_CELL + '\n' +
           'New row: Gemstone (' + PAID_GEM_CELL + ' / ' + GEM_VAL_CELL + ')\n\n' +
           'Old NET read: ' + beforeNet + '\n' +
           'Re-enter the order number in ' + ORDER_NUM_CELL + ' to repopulate, then check the new ' +
           'NET matches.');
}

// Creates the Voucher Log / Exchange Log tabs if missing (extracted so the fresh build can call it).
function ensureLogTabs_(ss) {
  if (!ss.getSheetByName(VOUCHER_LOG)) {
    var old = ss.getSheetByName('CN Log');
    if (old) old.setName(VOUCHER_LOG); else ss.insertSheet(VOUCHER_LOG);
  }
  if (!ss.getSheetByName(EXCHANGE_LOG)) {
    var ex = ss.insertSheet(EXCHANGE_LOG);
    ex.appendRow(['Issued', 'EXC Number', 'Old Order', 'New Draft', 'Customer', 'Email',
                  'Net Wt', 'Dia Wt', 'Gold Val', 'Dia Val', 'Exchange Value', 'Status', 'New Draft ID']);
  }
  ensureAfterSalesLog_(ss);
}

// ═════════════════════════════════════════════════════════════════════════════
// AFTER-SALES LOG — one tab, every after-sales event
//
// Voucher Log and Exchange Log each record ONE document type, and neither records WHY the document
// was raised: Source (Purchase Exchange / Old Gold), Exchange Type (Deduction / Full Value) and
// Document Type (Voucher / Exchange Note) are read off the control cells at issue time and then
// thrown away. Their Status column is also frozen at issue — it never learns that the voucher was
// later spent, expired or voided. And a gateway refund never touched this sheet at all.
//
// This tab is the single answer to "what did after-sales do, and what happened to it". The two old
// tabs are left exactly as they are: voidVoucher and voidExchangeNote locate rows by FIXED COLUMN
// INDEX (row[10] expiry, row[12] price_rule_id), so widening them would break the paths that delete
// live Shopify discounts. This one is written to in ADDITION, never instead.
//
// Rows are written by name, never by position. Voucher Log already shows what positional appends
// cost: column N holds the override reason on an old-gold row and the source on a repair-refund row.
// ═════════════════════════════════════════════════════════════════════════════

const AFTERSALES_LOG = 'After-Sales Log';

// The header IS the schema — logAfterSales_ maps a keyed object onto it, so adding a column here is
// the only change needed and no existing writer has to be touched.
const AFTERSALES_COLS = [
  'Logged At',      // when this row was written (dd-MM-yyyy HH:mm)
  'Event',          // issued | applied | voided | refunded
  'Source',         // Purchase Exchange | Old Gold | Repair refund | Gateway refund
  'Exchange Type',  // Deduction | Full Value | —
  'Doc Type',       // Voucher | Exchange Note | Refund
  'Store',
  'Serial',         // VCH.../EXC... — blank for a gateway refund, which has no serial
  'Order',          // the order the event arose from (or OLD-GOLD)
  'Applied To',     // draft/order the credit was spent on
  'Value',
  'Net Wt', 'Dia Cts', 'Gold Value', 'Diamond Value',
  'Customer', 'Email',
  'Expiry',
  'Status',         // refreshed from the ledger: outstanding | applied | redeemed | voided | expired | refunded
  'Redeemed On',    // the order it was TRULY redeemed against (from the ledger, not from issue time)
  'Reason / Note',  // override reason, refund reason, gateway name
  'Ref',            // idempotency key — refund id / draft ref / price_rule_id
  'Status Synced'   // when Status/Redeemed On were last refreshed from the ledger
];

function ensureAfterSalesLog_(ss) {
  var sh = ss.getSheetByName(AFTERSALES_LOG);
  if (!sh) {
    sh = ss.insertSheet(AFTERSALES_LOG);
    sh.appendRow(AFTERSALES_COLS);
    sh.getRange(1, 1, 1, AFTERSALES_COLS.length).setFontWeight('bold');
    sh.setFrozenRows(1);
  } else if (sh.getLastColumn() < AFTERSALES_COLS.length) {
    // A tab created by an older version of this script is widened in place rather than rebuilt —
    // rebuilding would drop the history this tab exists to hold.
    sh.getRange(1, 1, 1, AFTERSALES_COLS.length).setValues([AFTERSALES_COLS]).setFontWeight('bold');
  }
  return sh;
}

// Column index (1-based) of a header name. Throws on an unknown name so a typo fails loudly at the
// call site instead of silently writing into the wrong column.
function afterSalesCol_(header) {
  var i = AFTERSALES_COLS.indexOf(header);
  if (i === -1) throw new Error('After-Sales Log: unknown column "' + header + '"');
  return i + 1;
}

// Appends one event. NEVER THROWS — this is bookkeeping hung off flows that create real money
// instruments, and a logging hiccup must not roll back a voucher that Shopify has already issued.
// Failures go to Logger so they are recoverable, not silent.
function logAfterSales_(fields) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ensureAfterSalesLog_(ss);
    if (!fields['Logged At']) {
      fields['Logged At'] = Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), 'dd-MM-yyyy HH:mm');
    }
    sh.appendRow(AFTERSALES_COLS.map(function (h) {
      return fields[h] === undefined || fields[h] === null ? '' : fields[h];
    }));
    return true;
  } catch (e) {
    Logger.log('After-Sales Log write failed (' + (fields && fields.Serial) + '): ' + e.message);
    return false;
  }
}

// The three control-cell classifications every counter-raised document shares. Read once, here, so
// the four create flows cannot each interpret the dropdowns differently.
function afterSalesContext_(calc) {
  var src  = String(calc.getRange(SOURCE_CELL).getValue()).trim();
  var exch = String(calc.getRange(EXCHTYPE_CELL).getValue()).trim();
  return {
    'Source':        src || 'Purchase Exchange',
    // Old gold has no original invoice, so Deduction vs Full Value is meaningless for it — recording
    // whatever the dropdown happened to be left on would be noise in the report.
    'Exchange Type': src.toLowerCase() === SOURCE_OLD_GOLD ? '—' : exch,
    'Doc Type':      String(calc.getRange(DOCTYPE_CELL).getValue()).trim(),
    'Store':         resolveStoreCode_()
  };
}

// Every 'Ref' already in the log — the idempotency set for the refund sync and for any future
// replayable writer. Read once per sync rather than per candidate row.
function afterSalesRefs_(sh) {
  var seen = {};
  var last = sh.getLastRow();
  if (last < 2) return seen;
  var col = afterSalesCol_('Ref');
  sh.getRange(2, col, last - 1, 1).getValues().forEach(function (r) {
    var v = String(r[0]).trim();
    if (v) seen[v] = true;
  });
  return seen;
}

// ═════════════════════════════════════════════════════════════════════════════
// RETURNS — the human side of a return, folded into the consolidated tab
//
// The Returns Log is filled in by hand and covers BOTH kinds of refund: on a real order, and of an
// advance paid on a draft order. Both of those already reach the consolidated tab on their own —
// syncShopifyRefunds brings the first, syncDraftRefunds the second — but each brings only the MONEY.
// What the request was for, which platform it went back on, and where it stands are known to the
// staff member who typed them and to nothing else.
//
// So a refund ends up as TWO rows, deliberately: 'return requested' with no Value, and 'refunded'
// with the amount. They are two facts about one event and collapsing them would lose one of them —
// a request that was declined has no refund row at all, and a refund that nobody logged a request
// for is exactly the thing worth noticing. For a money total, filter Event = refunded (equivalently,
// Value non-blank); that set is untouched by anything written here.
//
// So this reads the Returns Log and writes each row across as a 'return' event. It deliberately
// writes NO Value: the request is not money, and the gateway sync already logs the amount when the
// refund lands. Putting a figure here as well would double-count every return in any total.
//
// Append-only and idempotent on Ref = 'return:<order>:<date>'. A row already synced is skipped, so
// re-running is free — but note that means a STATUS edited in the Returns Log afterwards does not
// propagate. Once a return is in the consolidated tab, update it there.
//
// RETURNS_LOG must match the tab name exactly. Correct it here if the tab is called something else.
// ═════════════════════════════════════════════════════════════════════════════

const RETURNS_LOG = 'Returns Log';

// Returns Log columns, as laid out by hand:
//   0 Order Date  1 Order No.  2 Customer Name  3 Customer Email  4 Request Type
//   5 Reason      6 Refund Platform            7 Status          8 Notes
function syncReturnsLog()          { syncReturnsLog_(true); }
function syncReturnsLogScheduled() { syncReturnsLog_(false); }

function syncReturnsLog_(showUi) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ui = showUi ? SpreadsheetApp.getUi() : null;
  var rl = ss.getSheetByName(RETURNS_LOG);
  if (!rl) {
    if (ui) ui.alert('Tab "' + RETURNS_LOG + '" not found.\n\nRename it, or change RETURNS_LOG at the ' +
                     'top of this section to match what the tab is actually called.');
    return { added: 0 };
  }

  var sh   = ensureAfterSalesLog_(ss);
  var seen = afterSalesRefs_(sh);
  var tz   = ss.getSpreadsheetTimeZone();
  var fmt  = function (d) {
    if (!d) return '';
    return (d instanceof Date) ? Utilities.formatDate(d, tz, 'dd-MM-yyyy') : String(d).trim();
  };

  var added = 0, skipped = 0;
  rl.getDataRange().getValues().slice(1).forEach(function (v) {
    var orderNo = String(v[1] || '').trim();
    if (!orderNo) return;                       // a blank row, or a note typed under the table
    var when = fmt(v[0]);
    var ref  = 'return:' + orderNo + ':' + when;
    if (seen[ref]) { skipped++; return; }

    // Which of the two kinds this is, read off the reference — a draft order is #D-prefixed. Worth
    // recording, because it is the one distinction the Returns Log makes that the tab could not
    // otherwise recover, and the two refund syncs already separate themselves the same way.
    var isDraft = /^#?D/i.test(orderNo);

    var note = [String(v[5] || '').trim(),
                v[6] ? 'via ' + String(v[6]).trim() : '',
                String(v[8] || '').trim()].filter(Boolean).join(' — ');

    logAfterSales_({
      'Logged At':     when,
      'Event':         'return requested',
      'Source':        isDraft ? 'Return — draft advance' : 'Return — order',
      'Exchange Type': '—',
      // The request type as staff classified it (return / exchange / repair), kept verbatim rather
      // than forced into the voucher/exchange vocabulary, which would lose the distinction.
      'Doc Type':      String(v[4] || 'Return').trim(),
      'Order':         orderNo,
      'Customer':      String(v[2] || '').trim(),
      'Email':         String(v[3] || '').trim(),
      'Status':        String(v[7] || '').trim() || 'open',
      'Reason / Note': note,
      'Ref':           ref
    });
    seen[ref] = true;
    added++;
  });

  if (ui) {
    ui.alert('✅ Returns synced.\n\n' +
             'New rows added: ' + added + '\n' +
             'Already present: ' + skipped + '\n\n' +
             'Each refund now shows TWO rows: "return requested" (no Value) from here, and "refunded" ' +
             '(with the amount) from the refund syncs. That is deliberate — a declined request has no ' +
             'refund row, and a refund with no request is worth spotting.\n\n' +
             'For a money total, filter Event = refunded. Nothing written here carries a Value, so it ' +
             'cannot inflate one.');
  }
  return { added: added, skipped: skipped };
}

// ═════════════════════════════════════════════════════════════════════════════
// BACKFILL — bring the old tabs into the After-Sales Log
//
// The After-Sales Log only holds what has been raised since it was created, so every voucher and
// exchange note issued before that exists solely in Voucher Log / Exchange Log. Any report built on
// the consolidated tab therefore understates everything before that date. This reads both old tabs
// and writes across whatever is missing.
//
// DRY RUN BY DEFAULT — it counts and shows a sample, and writes nothing until you run the WRITE
// version. Idempotent either way: every row is keyed Ref = 'backfill:<serial>' and afterSalesRefs_
// already dedupes on Ref, so a second run adds nothing.
//
// It does NOT touch Voucher Log or Exchange Log. voidVoucher and voidExchangeNote find their rows by
// FIXED COLUMN INDEX in those tabs, so they must stay exactly as they are.
//
// Status comes across frozen at whatever the old tab recorded at issue. Run "Refresh log statuses
// from ledger" straight afterwards to replace it with what actually happened — serials the ledger
// does not know are left alone and reported, rather than guessed at.
// ═════════════════════════════════════════════════════════════════════════════

function backfillAfterSalesLog()      { backfillAfterSalesLog_(false); }
function backfillAfterSalesLogApply() { backfillAfterSalesLog_(true); }

function backfillAfterSalesLog_(apply) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ui = SpreadsheetApp.getUi();
  var sh = ensureAfterSalesLog_(ss);
  var seen = afterSalesRefs_(sh);

  var rows = []
    .concat(backfillRowsFromVoucherLog_(ss))
    .concat(backfillRowsFromExchangeLog_(ss))
    .filter(function (r) { return r && r.Serial && !seen[r.Ref]; });

  if (!rows.length) {
    ui.alert('Nothing to backfill.\n\nEvery voucher and exchange note in the old tabs is already in ' +
             AFTERSALES_LOG + ', or the old tabs are empty.');
    return;
  }

  if (!apply) {
    var sample = rows.slice(0, 8).map(function (r) {
      return '  ' + String(r['Doc Type']).padEnd(14) + String(r.Serial).padEnd(22) +
             String(r.Order || '').padEnd(12) + (r.Value || '');
    }).join('\n');
    ui.alert('DRY RUN — nothing written.\n\n' +
             rows.length + ' row(s) would be added to ' + AFTERSALES_LOG + '.\n\n' +
             sample + (rows.length > 8 ? '\n  … and ' + (rows.length - 8) + ' more' : '') +
             '\n\nRun "Backfill After-Sales Log (WRITE)" to apply, then "Refresh log statuses from ' +
             'ledger" to correct the statuses.');
    return;
  }

  rows.forEach(function (r) { logAfterSales_(r); });
  ui.alert('✅ Backfill complete.\n\n' + rows.length + ' row(s) added to ' + AFTERSALES_LOG + '.\n\n' +
           'Old tabs untouched. Now run "Refresh log statuses from ledger" — the statuses just written ' +
           'are frozen at issue time and the ledger knows what actually happened since.');
}

// Voucher Log columns, as written by createVoucher_ / createOldGoldVoucher_ / handleIssueVoucher_:
//   0 issued  1 serial  2 order  3 customer  4 email  5 netWt  6 diaCts  7 goldVal  8 diaVal
//   9 value  10 expiry  11 status  12 price_rule_id  13 reason-or-source  14 repair draft ref
//
// Column 13 is overloaded — the override reason on an old-gold row, the source on a repair-refund
// row — so the source is inferred from the row's SHAPE rather than from that cell: a repair refund is
// the only one carrying a draft ref in 14, and old gold is the only one with 'OLD-GOLD' in 2.
function backfillRowsFromVoucherLog_(ss) {
  var log = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
  if (!log || log.getLastRow() < 2) return [];
  return log.getDataRange().getValues().slice(1).map(function (v) {
    var serial = String(v[1] || '').trim();
    if (!serial) return null;
    var order    = String(v[2] || '').trim();
    var draftRef = String(v[14] || '').trim();
    var source   = draftRef ? 'Repair refund'
                 : (order.toUpperCase() === 'OLD-GOLD' ? 'Old Gold' : 'Purchase Exchange');
    return {
      'Logged At': v[0], 'Event': 'issued',
      'Source': source,
      // Not recorded on the old tabs at all. Blank, never guessed — a confidently wrong
      // classification is worse in a report than an obviously missing one.
      'Exchange Type': '—', 'Store': '',
      'Doc Type': 'Voucher', 'Serial': serial, 'Order': order,
      'Applied To': draftRef,
      'Value': v[9], 'Net Wt': v[5], 'Dia Cts': v[6], 'Gold Value': v[7], 'Diamond Value': v[8],
      'Customer': v[3], 'Email': v[4], 'Expiry': v[10],
      'Status': v[11] || 'outstanding',
      'Reason / Note': String(v[13] || '').trim(),
      'Ref': 'backfill:' + serial
    };
  });
}

// Exchange Log columns, as written by createExchangeNote_:
//   0 issued  1 serial  2 old order  3 new draft  4 customer  5 email  6 netWt  7 diaCts
//   8 goldVal  9 diaVal  10 value  11 status  12 new draft id
function backfillRowsFromExchangeLog_(ss) {
  var log = ss.getSheetByName(EXCHANGE_LOG);
  if (!log || log.getLastRow() < 2) return [];
  return log.getDataRange().getValues().slice(1).map(function (v) {
    var serial = String(v[1] || '').trim();
    if (!serial) return null;
    return {
      'Logged At': v[0], 'Event': 'issued',
      'Source': 'Purchase Exchange', 'Exchange Type': '—', 'Store': '',
      'Doc Type': 'Exchange Note', 'Serial': serial, 'Order': String(v[2] || '').trim(),
      'Applied To': String(v[3] || '').trim(),
      'Value': v[10], 'Net Wt': v[6], 'Dia Cts': v[7], 'Gold Value': v[8], 'Diamond Value': v[9],
      'Customer': v[4], 'Email': v[5],
      // An exchange note is deducted on the spot, so 'applied' is its state from birth.
      'Status': v[11] || 'applied',
      'Ref': 'backfill:' + serial
    };
  });
}

// ═════════════════════════════════════════════════════════════════════════════
// REFUND SYNC — the one after-sales event this sheet does not raise
//
// A refund is initiated at the gateway and only RECORDED on Shopify; nothing about it passes
// through this sheet, so it cannot be logged at the point of action like everything else. This
// pulls them instead: walk orders touched since the last run, read order.refunds[], append the ones
// not already logged.
//
// Pull, not webhook, on purpose. A webhook that fails or is never delivered loses that refund
// permanently and silently; this re-reads a window every hour, so a failed run costs nothing but an
// hour's delay. The cursor is deliberately rewound by REFUND_SYNC_OVERLAP_MIN before querying,
// because Shopify's updated_at is server-side and an order updated during a run can otherwise fall
// between two windows. Re-reading a refund is free — the Ref check drops it.
// ═════════════════════════════════════════════════════════════════════════════

const REFUND_CURSOR_KEY        = 'refund_sync_cursor';   // ISO timestamp in Script Properties
const REFUND_SYNC_OVERLAP_MIN  = 30;                     // rewind on each run, guards the window seam
const REFUND_SYNC_FIRST_RUN_D  = 30;                     // how far back a never-run sync reaches

function syncShopifyRefunds() { return syncShopifyRefunds_(true); }

// silent=false when called from the hourly trigger (no UI scope there).
function syncShopifyRefunds_(showUi) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sh    = ensureAfterSalesLog_(ss);
  var props = PropertiesService.getScriptProperties();
  var tz    = ss.getSpreadsheetTimeZone();

  var cursor = props.getProperty(REFUND_CURSOR_KEY);
  var since  = cursor ? new Date(cursor) : new Date(Date.now() - REFUND_SYNC_FIRST_RUN_D * 864e5);
  since = new Date(since.getTime() - REFUND_SYNC_OVERLAP_MIN * 60000);

  // Stamped BEFORE the fetch. An order updated while this run is in flight must land in the next
  // window, not be skipped by a cursor stamped after the fact.
  var runStartedAt = new Date();

  var seen    = afterSalesRefs_(sh);
  var added   = 0;
  var scanned = 0;
  var pageInfo = null;
  var guard    = 0;

  try {
    do {
      var endpoint = pageInfo
        ? 'orders.json?limit=250&page_info=' + encodeURIComponent(pageInfo)
        : 'orders.json?limit=250&status=any&updated_at_min=' + encodeURIComponent(since.toISOString()) +
          '&fields=id,name,email,customer,refunds,created_at,updated_at';
      var page = shopifyGetPaged_(endpoint);
      if (!page || !page.body || !page.body.orders) break;

      page.body.orders.forEach(function (order) {
        scanned++;
        (order.refunds || []).forEach(function (refund) {
          var ref = 'refund:' + refund.id;
          if (seen[ref]) return;

          var amount = refundAmount_(refund);
          // A zero-value refund is a restock or an adjustment-only record, not money returned.
          // Logging it would inflate every refund total in the report.
          if (!(amount > 0)) return;

          var cust = order.customer || {};
          logAfterSales_({
            'Logged At':     Utilities.formatDate(new Date(refund.created_at || order.updated_at), tz, 'dd-MM-yyyy HH:mm'),
            'Event':         'refunded',
            'Source':        'Gateway refund',
            'Exchange Type': '—',
            'Doc Type':      'Refund',
            // Store is deliberately BLANK. resolveStoreCode_() reads whatever the calculator's
            // control cell happens to be set to right now, which has nothing to do with where the
            // refunded order was placed — stamping it would put a confidently wrong store on every
            // refund row. Blank is recoverable; wrong is not.
            'Order':         order.name || ('#' + order.id),
            'Value':         amount,
            'Customer':      [cust.first_name, cust.last_name].filter(Boolean).join(' '),
            'Email':         order.email || cust.email || '',
            'Status':        'refunded',
            'Reason / Note': refundNote_(refund),
            'Ref':           ref,
            'Status Synced': Utilities.formatDate(new Date(), tz, 'dd-MM-yyyy HH:mm')
          });
          seen[ref] = true;
          added++;
        });
      });

      pageInfo = page.nextPageInfo;
    } while (pageInfo && ++guard < 40);
  } catch (e) {
    // Cursor is NOT advanced on failure, so the next run re-covers this window.
    Logger.log('refund sync failed: ' + e.message);
    if (showUi) SpreadsheetApp.getUi().alert('❌ Refund sync failed:\n\n' + e.message +
                                             '\n\nThe cursor was not advanced — the next run retries this window.');
    return { added: 0, scanned: scanned, error: e.message };
  }

  props.setProperty(REFUND_CURSOR_KEY, runStartedAt.toISOString());

  if (showUi) {
    SpreadsheetApp.getUi().alert('✅ Refund sync complete.\n\n' +
      'Orders scanned: ' + scanned + '\n' +
      'New refunds logged: ' + added + '\n' +
      'Window: since ' + Utilities.formatDate(since, tz, 'dd-MM-yyyy HH:mm') + '\n\n' +
      'Next run reads from ' + Utilities.formatDate(runStartedAt, tz, 'dd-MM-yyyy HH:mm') + '.');
  }
  return { added: added, scanned: scanned };
}

// DRAFT-ORDER refunds — the other half of the refund picture.
//
// syncShopifyRefunds_ above reads refunds off ORDERS, because that is where Shopify keeps them. A
// DRAFT has no refund object at all: when a deposit on a draft goes back (the sale fell through, or
// the order value contracted) Shopify records nothing, so nothing above can ever see it. Those
// refunds are recorded by the middleware into credit_instruments as instrument_type 'refund', and
// this pulls them into the same log so all after-sales sits on one platform.
//
// A PULL, deliberately, for the reason written at the top of the refund-sync block: a push that
// fails or is never delivered loses the refund forever. Re-running this is always safe.
//
// No new columns. The 22-column schema already carries these rows — 'refunded', 'Refund' and the
// 'refunded' status are existing values. The only new value anywhere is Source 'Draft refund',
// alongside 'Gateway refund' and 'Repair refund'.
//
// Unlike a gateway refund these rows DO carry a Serial ('#D189-R1'), so refreshLogStatuses() will
// try to join them to the ledger. That is harmless — the ledger returns 'refunded', which is what
// the row already says — but it is worth watching on the first run.
function syncDraftRefunds_(showUi) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ensureAfterSalesLog_(ss);
  var tz = ss.getSpreadsheetTimeZone();

  var res = reconLedgerView_('detail', 'type=refund');
  if (!res.ok) {
    Logger.log('draft refund sync failed: ' + res.error);
    if (showUi) SpreadsheetApp.getUi().alert('❌ Draft refund sync failed:\n\n' + res.error +
                                             '\n\nNothing was written. Run it again once the middleware responds.');
    return { added: 0, scanned: 0, error: res.error };
  }

  // Read once per run, not per candidate — same as the gateway refund sync.
  var seen    = afterSalesRefs_(sh);
  var added   = 0;
  var scanned = 0;

  res.rows.forEach(function (r) {
    if (String(r.instrument_type) !== 'refund') return;   // belt-and-braces if type= is ever ignored
    scanned++;
    var ref = 'draft-refund:' + r.serial_code;
    if (seen[ref]) return;

    // Mirrors the zero-value guard on the gateway side: a row with nothing behind it would inflate
    // every refund total in the log.
    var amount = toNum(r.value);
    if (!(amount > 0)) return;

    var note = [];
    if (r.refund_mode) note.push('via ' + r.refund_mode);
    if (r.gateway_ref) note.push('UTR ' + r.gateway_ref);
    if (r.voided_at)   note.push('draft deleted');

    logAfterSales_({
      'Logged At':     Utilities.formatDate(r.refunded_at ? new Date(r.refunded_at) : new Date(), tz, 'dd-MM-yyyy HH:mm'),
      'Event':         'refunded',
      'Source':        'Draft refund',
      'Exchange Type': '—',
      'Doc Type':      'Refund',
      // Unlike the gateway sync, the store IS known here: the middleware stamps it from the
      // document's own custom.state_code, so it is the store the sale was actually raised at rather
      // than whatever the calculator's control cell happens to be set to right now.
      'Store':         r.state_code || '',
      'Serial':        r.serial_code || '',
      'Order':         r.issued_against || '',
      'Value':         amount,
      'Customer':      r.customer_name || '',
      'Status':        'refunded',
      'Reason / Note': note.join(' · '),
      'Ref':           ref,
      'Status Synced': Utilities.formatDate(new Date(), tz, 'dd-MM-yyyy HH:mm')
    });
    seen[ref] = true;
    added++;
  });

  if (showUi) {
    SpreadsheetApp.getUi().alert('✅ Draft refund sync complete.\n\n' +
      'Refunds seen: ' + scanned + '\n' +
      'New rows logged: ' + added + '\n\n' +
      'Rows already in the log are skipped by their Ref, so this is safe to run any time.');
  }
  return { added: added, scanned: scanned };
}

// Trigger entry point — no UI scope, so it must not try to alert.
function syncDraftRefundsScheduled() { syncDraftRefunds_(false); }

// Menu wrapper, matching syncShopifyRefunds.
function syncDraftRefunds() { syncDraftRefunds_(true); }

// Money actually returned = the sum of the refund's transactions. refund.transactions is the only
// field that reflects what the GATEWAY moved; the line-item subtotals include restocks that never
// resulted in a payment reversal.
function refundAmount_(refund) {
  var total = 0;
  (refund.transactions || []).forEach(function (t) {
    if (String(t.kind) === 'refund' && String(t.status) === 'success') total += toNum(t.amount);
  });
  return Math.round(total * 100) / 100;
}

// Gateway + staff note, for the report's "why". Falls back to the line-item titles when there is no
// note, so a bare refund row still says what went back.
function refundNote_(refund) {
  var parts = [];
  var gateways = {};
  (refund.transactions || []).forEach(function (t) { if (t.gateway) gateways[t.gateway] = true; });
  var gw = Object.keys(gateways);
  if (gw.length) parts.push(gw.join(', '));
  if (refund.note) parts.push(String(refund.note).trim());
  if (!refund.note) {
    var items = (refund.refund_line_items || [])
      .map(function (rli) { return rli.line_item && (rli.line_item.sku || rli.line_item.title); })
      .filter(Boolean);
    if (items.length) parts.push(items.join(' | '));
  }
  return parts.join(' — ');
}

// Cursor-paginated GET. shopifyGet returns only the body, but Shopify pages orders through the Link
// HEADER, so the plain helper cannot walk past the first 250 — on a busy month that silently drops
// every refund after the first page.
function shopifyGetPaged_(endpoint) {
  var res = UrlFetchApp.fetch('https://' + SHOPIFY_SHOP + '/admin/api/2024-01/' + endpoint, {
    method: 'get',
    headers: { 'X-Shopify-Access-Token': getToken(), 'Content-Type': 'application/json' },
    muteHttpExceptions: true
  });
  if (res.getResponseCode() !== 200) {
    throw new Error('Shopify ' + res.getResponseCode() + ': ' + res.getContentText().substring(0, 200));
  }
  var link = res.getHeaders()['Link'] || res.getHeaders()['link'] || '';
  var next = /<[^>]*[?&]page_info=([^&>]+)[^>]*>;\s*rel="next"/.exec(link);
  return { body: JSON.parse(res.getContentText()), nextPageInfo: next ? decodeURIComponent(next[1]) : null };
}

// ═════════════════════════════════════════════════════════════════════════════
// STATUS REFRESH — teach the log what happened AFTER issue
//
// The sheet only ever knows the moment of issue. The lifecycle — applied to a draft, truly redeemed
// on an order, reopened, voided, expired — is tracked in credit_instruments and already served by
// /api/recon-ledger?view=detail. This pulls that back onto the log rows by serial, so "was it
// redeemed, and against what" is answerable from the tab instead of from the database.
// ═════════════════════════════════════════════════════════════════════════════

function refreshLogStatuses() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ensureAfterSalesLog_(ss);
  var ui = SpreadsheetApp.getUi();
  var tz = ss.getSpreadsheetTimeZone();

  var ledger;
  try {
    ledger = fetchLedgerDetail_();
  } catch (e) {
    ui.alert('❌ Could not read the ledger:\n\n' + e.message +
             '\n\nStatuses are unchanged — nothing was overwritten with a guess.');
    return;
  }

  var last = sh.getLastRow();
  if (last < 2) { ui.alert('Nothing logged yet.'); return; }

  var serialCol   = afterSalesCol_('Serial');
  var statusCol   = afterSalesCol_('Status');
  var redeemedCol = afterSalesCol_('Redeemed On');
  var appliedCol  = afterSalesCol_('Applied To');
  var syncedCol   = afterSalesCol_('Status Synced');

  var serials = sh.getRange(2, serialCol, last - 1, 1).getValues();
  var status  = sh.getRange(2, statusCol,   last - 1, 1).getValues();
  var redeem  = sh.getRange(2, redeemedCol, last - 1, 1).getValues();
  var applied = sh.getRange(2, appliedCol,  last - 1, 1).getValues();
  var synced  = sh.getRange(2, syncedCol,   last - 1, 1).getValues();
  var stamp   = Utilities.formatDate(new Date(), tz, 'dd-MM-yyyy HH:mm');
  var updated = 0, unknown = 0;

  for (var i = 0; i < serials.length; i++) {
    var code = String(serials[i][0]).trim().toUpperCase();
    if (!code) continue;                       // refund rows carry no serial — their status is final
    var row = ledger[code];
    if (!row) { unknown++; continue; }          // in the sheet but not the ledger — leave it alone and report it
    if (String(status[i][0]) === row.state && String(redeem[i][0]) === row.redeemed_against) continue;
    status[i][0]  = row.state;
    redeem[i][0]  = row.redeemed_against;
    if (row.applied_to_draft) applied[i][0] = row.applied_to_draft;
    synced[i][0]  = stamp;
    updated++;
  }

  sh.getRange(2, statusCol,   last - 1, 1).setValues(status);
  sh.getRange(2, redeemedCol, last - 1, 1).setValues(redeem);
  sh.getRange(2, appliedCol,  last - 1, 1).setValues(applied);
  sh.getRange(2, syncedCol,   last - 1, 1).setValues(synced);

  ui.alert('✅ Statuses refreshed from the ledger.\n\n' +
           'Rows updated: ' + updated + '\n' +
           (unknown ? 'Serials not found in the ledger: ' + unknown +
                      '\n(left untouched — these were issued before the ledger, or the ledger write failed)\n' : '') +
           '\nStatus values: outstanding | applied | redeemed | voided | expired' +
           (ledger.__partial
             ? '\n\n⚠️ PARTIAL REFRESH — this middleware build has no "detail" view, so only OPEN and ' +
               'REDEEMED instruments could be read. Applied / voided / expired ones are among the ' +
               (unknown ? unknown + ' ' : '') + 'not-found serials above and kept their existing status. ' +
               'Deploy the detail view for a full refresh.'
             : ''));
}

// serial_code → { state, redeemed_against, applied_to_draft } for every instrument in the ledger.
//
// TWO middleware versions are in the wild and this must work against both:
//   • view=detail        — one row per instrument, every state. The restructured build.
//   • outstanding+tieout — the only views the CURRENT deployment has; it 400s on anything else.
//
// So: try detail, and fall back to composing the same map from the two older views. The fallback is
// LOSSY and says so — it can only see open and redeemed instruments, because the deployed build has
// no view that lists applied/voided/expired ones. Those serials come back absent, and
// refreshLogStatuses leaves absent serials untouched rather than guessing, so a voided voucher keeps
// the 'voided' status its own void event already wrote. The moment the detail view deploys, this
// silently upgrades to full fidelity with no change here.
function fetchLedgerDetail_() {
  var detail = reconLedgerView_('detail');
  if (detail.ok) {
    var out = {};
    detail.rows.forEach(function (r) {
      if (!r.serial_code) return;
      out[String(r.serial_code).trim().toUpperCase()] = {
        state:            r.state || '',
        redeemed_against: r.redeemed_against || '',
        applied_to_draft: r.applied_to_draft || ''
      };
    });
    return out;
  }
  // 400 = this deployment predates the detail view. Any other failure is a real outage worth raising.
  if (detail.code !== 400) throw new Error(detail.error);

  var open   = reconLedgerView_('outstanding');
  var redeem = reconLedgerView_('tieout');
  if (!open.ok)   throw new Error(open.error);
  if (!redeem.ok) throw new Error(redeem.error);

  var merged = {};
  open.rows.forEach(function (r) {
    if (!r.serial_code) return;
    merged[String(r.serial_code).trim().toUpperCase()] =
      { state: 'outstanding', redeemed_against: '', applied_to_draft: '' };
  });
  // Redeemed wins over outstanding — an instrument cannot be both, and tieout is the later truth.
  redeem.rows.forEach(function (r) {
    if (!r.serial_code) return;
    merged[String(r.serial_code).trim().toUpperCase()] = {
      state:            'redeemed',
      redeemed_against: r.target_order_name || '',
      applied_to_draft: r.target_draft_id || ''
    };
  });
  merged.__partial = true;   // read by refreshLogStatuses to warn instead of implying full coverage
  return merged;
}

// One recon-ledger view. Never throws — returns {ok, rows, code, error} so the caller can tell a
// missing VIEW (400, expected on the current deployment) from a middleware outage.
// extraQs is an optional already-encoded query fragment (e.g. 'type=refund'). Optional so every
// existing single-argument caller is untouched.
function reconLedgerView_(view, extraQs) {
  try {
    var url  = MIDDLEWARE_URL + '/api/recon-ledger?view=' + view + (extraQs ? '&' + extraQs : '');
    var res  = UrlFetchApp.fetch(url, {
      method: 'get', muteHttpExceptions: true
    });
    var code = res.getResponseCode();
    if (code !== 200) {
      return { ok: false, code: code, rows: [],
               error: 'middleware ' + code + ' on view=' + view + ' — ' + res.getContentText().substring(0, 200) };
    }
    var body = JSON.parse(res.getContentText());
    return { ok: true, code: 200, rows: body.rows || body.data || (Array.isArray(body) ? body : []) };
  } catch (e) {
    return { ok: false, code: 0, rows: [], error: 'view=' + view + ': ' + e.message };
  }
}

// Installs the hourly refund pull. Separate from setupTriggers() (which owns the onEdit trigger)
// so re-running one never silently removes the other.
function setupRefundSync() {
  ScriptApp.getProjectTriggers()
    .filter(function (t) {
      var f = t.getHandlerFunction();
      return f === 'syncShopifyRefundsScheduled' || f === 'syncDraftRefundsScheduled';
    })
    .forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('syncShopifyRefundsScheduled').timeBased().everyHours(1).create();
  // Draft refunds ride the same hourly cadence. Two triggers rather than one function calling both,
  // so a failure in one pull cannot stop the other from running.
  ScriptApp.newTrigger('syncDraftRefundsScheduled').timeBased().everyHours(1).create();
  SpreadsheetApp.getUi().alert('✅ Hourly refund sync installed.\n\n' +
    'It reads Shopify order refunds AND draft-order refunds into "' + AFTERSALES_LOG + '". ' +
    'Run "Sync Shopify refunds now" or "Sync draft refunds now" any time to pull immediately.');
}

function restructureControlsToTop() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  // Guard: the label to the left of SOURCE_CELL is 'Source' only after a successful migration.
  var srcLabel = String(calc.getRange(cellRow_(SOURCE_CELL), 1).getValue()).trim();
  if (srcLabel === 'Source') { ui.alert('Already restructured — controls are at the top. Nothing changed.'); return; }

  var confirm = ui.alert('Restructure the Exchange Calculator?',
    'This inserts a CONTROLS block at the top and shifts existing rows down by ' + RESTRUCTURE_ROWS + '.\n\n' +
    'Run this on a COPY of the sheet first and verify the calc numbers before using it live.\n\nProceed?',
    ui.ButtonSet.YES_NO);
  if (confirm !== ui.Button.YES) return;

  // 1. Insert the top block. Sheets auto-adjusts all calc formulas for the downward shift.
  calc.insertRowsBefore(1, RESTRUCTURE_ROWS);

  // 2. Build the control fields. Label goes in column A of the same row as the value cell.
  function put(cellA1, label, opts) {
    var r = calc.getRange(cellA1);
    calc.getRange(r.getRow(), 1).setValue(label);
    if (opts && opts.list) {
      r.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(opts.list, true).setAllowInvalid(false).build());
      if (opts.def && !String(r.getValue()).trim()) r.setValue(opts.def);
    }
    if (opts && opts.note) r.setNote(opts.note);
  }
  calc.getRange('A1').setValue('0 — CONTROLS  (set these first)');
  put(SOURCE_CELL,        'Source',        { list: ['Purchase Exchange', 'Old Gold'], def: 'Purchase Exchange' });
  put(EXCHTYPE_CELL,      'Exchange Type', { list: ['Deduction', 'Full Value'],       def: 'Deduction' });
  put(DOCTYPE_CELL,       'Document Type', { list: ['Voucher', 'Exchange Note'],       def: 'Voucher' });
  put(STORE_CODE_CELL,    'Store Code',    { note: 'Issuing store, e.g. KA-HSR or KA-TEST. Blank = ' + STORE_CODE });
  refreshNewDraftField_(calc);   // label + note + clears any legacy gray-out rule
  calc.getRange(cellRow_(OG_CUSTOMER_CELL) - 1, 1).setValue('— OLD GOLD  (only when Source = Old Gold) —');
  put(OG_CUSTOMER_CELL,     'Old Gold — Customer phone/email', { note: 'Then run "Look up Old-Gold customer" (or just type — it auto-resolves).' });
  put(OG_CUSTID_CELL,       'Old Gold — Customer (auto)');
  put(OG_WEIGHT_CELL,       'Old Gold — Weight (g)');
  put(OG_PURITY_CELL,       'Old Gold — Purity (karat)');
  put(OG_RATE_CELL,         'Old Gold — Buy-back rate/g (auto)');
  put(OVERRIDE_REASON_CELL, 'Final value override — reason');

  // 3. Clear the legacy control cells (old rows 45-55, now shifted down by RESTRUCTURE_ROWS).
  for (var rr = 45 + RESTRUCTURE_ROWS; rr <= 55 + RESTRUCTURE_ROWS; rr++) {
    calc.getRange(rr, 1).clearContent();
    calc.getRange(rr, 2).clearContent();
    calc.getRange(rr, 2).clearDataValidations();
  }

  // 4. Gate the calc cells (warn-on-edit) at their new positions.
  lockCalcCells_(calc);

  ui.alert('✅ Restructured.\n\n' +
           'Controls are now at the top (rows 1-' + (RESTRUCTURE_ROWS - 1) + '). Existing rows moved down by ' +
           RESTRUCTURE_ROWS + ' and the calc formulas auto-adjusted.\n\n' +
           'Verify the NET value matches what it was before, then use as normal. ' +
           'Only the final value (' + NET_VALUE_CELL + ') is freely editable.');
}

// Warn-on-edit protection on the auto-calculated cells so staff can't hand-edit weights/rates/values.
// Warning-only works without domain-admin rights (a hard lock needs setDomainEdit the owner may lack).
function lockCalcCells_(calc) {
  var lockCells = [NET_WT_CELL, DIA_CTS_CELL, GOLD_RATE_ORD_CELL, GOLD_RATE_EFF_CELL,
                   LGD_RATE_CELL, DIA_VALUE_CELL, DIA_PAID_CELL, GOLD_VAL_CELL, DIA_VAL_CELL,
                   OG_RATE_CELL, OG_CUSTID_CELL]
    // The whole paid column is read off the order — hand-editing it would make the sheet claim the
    // customer was charged something they were not.
    .concat(CALC_ROWS.map(function (r) { return r.paid; }))
    // Every exchange-value row EXCEPT custom deductions, which is the one staff are meant to fill in.
    .concat(CALC_ROWS.filter(function (r) { return r.value !== CUSTOM_DED_CELL; })
                     .map(function (r) { return r.value; }));
  lockCells.forEach(function (a1) {
    var rng = calc.getRange(a1);
    calc.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(function (p) {
      if (p.getRange().getA1Notation() === rng.getA1Notation()) p.remove();
    });
    calc.getRange(a1).protect().setDescription('Auto-calculated — do not edit (Timanti CN)').setWarningOnly(true);
  });
}

// ── MIGRATION: collapse the duplicated gold-rate cells (C33 + D33) onto C33 ───────
// A live sheet carries the rate twice: C33 for display, D33 as the base of the gold-value formula.
// This moves the number onto C33, re-points the formula, and clears D33.
//
// ORDER MATTERS. The formula is re-pointed BEFORE D33 is cleared. Do it the other way and a run
// that dies in between leaves =B29*D33 pointing at an empty cell — gold valued at ZERO, with no
// error anywhere. Idempotent: re-running on an already-migrated sheet is a no-op.
function migrateGoldRateCell() {
  var ss   = SpreadsheetApp.getActiveSpreadsheet();
  var calc = ss.getSheetByName(CALC_SHEET_NAME);
  var ui   = SpreadsheetApp.getUi();
  if (!calc) { ui.alert('Sheet "' + CALC_SHEET_NAME + '" not found.'); return; }

  var legacy    = calc.getRange(GOLD_RATE_D33_LEGACY);
  var eff       = calc.getRange(GOLD_RATE_EFF_CELL);
  var legacyVal = legacy.getValue();
  var shown     = String(eff.getValue()).trim();
  var moved     = false;

  // D33 holds the authoritative number today. C33 holds either the same number (single SKU) or
  // the per-SKU display text (multi) — the text is preserved in the note, not thrown away.
  if (typeof legacyVal === 'number' && legacyVal > 0) {
    eff.setValue(legacyVal);
    eff.setNote(goldRateNote_(shown.indexOf('/') > -1, shown.indexOf('/') > -1 ? shown.split('/').map(function (s) { return s.trim(); }) : null));
    moved = true;
  }

  // Re-point first, clear second.
  calc.getRange(GOLD_VAL_CELL).setFormula('=' + NET_WT_CELL + '*' + GOLD_RATE_EFF_CELL);
  SpreadsheetApp.flush();

  legacy.clearContent();
  legacy.clearNote();
  // Drop the protection pinned to D33, or staff get "do not edit" warnings on a retired blank cell.
  calc.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(function (p) {
    if (p.getRange().getA1Notation() === legacy.getA1Notation()) p.remove();
  });

  calc.getRange(cellRow_(GOLD_RATE_ORD_CELL), 1).setValue(GOLD_RATE_LABEL);
  lockCalcCells_(calc);

  ui.alert('✅ Gold rate cells collapsed onto ' + GOLD_RATE_EFF_CELL + '.\n\n' +
           (moved ? 'Rate moved from ' + GOLD_RATE_D33_LEGACY + ': ' + legacyVal + '\n'
                  : 'Nothing to move — ' + GOLD_RATE_D33_LEGACY + ' was already empty.\n') +
           GOLD_VAL_CELL + ' now reads =' + NET_WT_CELL + '*' + GOLD_RATE_EFF_CELL + '.\n' +
           GOLD_RATE_D33_LEGACY + ' is retired and cleared.\n\n' +
           'Check that ' + GOLD_VAL_CELL + ' still shows the gold value it showed before.');
}

// ── EXCHANGE NOTE — instant post-tax deduction applied to a NEW invoice ──────────
// Staff ring up the new item (creating a Shopify draft), then enter that draft # in NEWDRAFT_CELL.
// The middleware appends a negative custom line item (EXC-...) to that draft.
function createExchangeNote_() {
  const ss   = SpreadsheetApp.getActiveSpreadsheet();
  const calc = ss.getSheetByName(CALC_SHEET_NAME);
  const log  = ss.getSheetByName(EXCHANGE_LOG);
  const ui   = SpreadsheetApp.getUi();
  if (!log) { ui.alert('Log tab "' + EXCHANGE_LOG + '" not found. Run "Set up Document Type fields" first.'); return; }

  const customerName  = calc.getRange(CUST_NAME_CELL).getValue();
  const customerEmail = calc.getRange(CUST_EMAIL_CELL).getValue();
  const orderNumber   = String(calc.getRange(ORDER_NUM_CELL).getValue()).trim();   // OLD order (item exchanged)
  const newDraftRef   = String(calc.getRange(NEWDRAFT_CELL).getValue()).trim();  // NEW sale's draft
  const netWt         = toNum(calc.getRange(NET_WT_CELL).getValue());
  const diaWt         = toNum(calc.getRange(DIA_CTS_CELL).getValue());
  const goldVal       = toNum(calc.getRange(GOLD_VAL_CELL).getValue());
  const diaVal        = toNum(calc.getRange(DIA_VAL_CELL).getValue());
  // The Value-to-Issue cell already reflects Deduction vs Full Value (maintained by
  // refreshFinalIssueValue_ / the dropdown onEdit). Refresh once more here so a stale sheet can't
  // issue the wrong figure, then read it.
  refreshFinalIssueValue_(calc);
  const excValue = toNum(calc.getRange(FINAL_ISSUE_CELL).getValue());
  if (!fullValueGuardPasses_(calc, ui, orderNumber, excValue)) return;

  if (!customerEmail || !orderNumber || excValue <= 0) {
    ui.alert('Missing data. Fill customer email, old order number, and ensure exchange value > 0.');
    return;
  }
  if (!newDraftRef) {
    ui.alert('Enter the new sale\'s draft order number in ' + NEWDRAFT_CELL + ' (e.g. #D123).\n\n' +
             'Ring up the new item as a draft in Shopify first, then run this again.');
    return;
  }

  const today  = new Date();
  // No local-format fallback here: an Exchange Note must not hit an invoice without a real
  // ledger number, so a failed allocation aborts rather than inventing one.
  const alloc  = allocateExcSerial();
  if (!alloc || !alloc.serial_code) { ui.alert('Could not allocate an Exchange Note number (middleware unreachable). Try again.'); return; }
  const excNum = alloc.serial_code;
  const issued = Utilities.formatDate(today, 'Asia/Kolkata', 'dd-MM-yyyy');

  // Apply the deduction to the new draft via the middleware. Returns the resolved numeric draft id.
  const result = applyExchangeNote_(newDraftRef, excNum, excValue, orderNumber, customerName);
  if (!result || !result.success) {
    ui.alert('❌ Exchange Note not applied.\n\n' +
             (result && result.error ? result.error : 'Middleware error — check that the draft number is correct.') +
             '\n\nThe number ' + excNum + ' was reserved; void it if you do not retry.');
    return;
  }
  const newDraftId   = result.draftId || '';
  const newDraftName = newDraftRef.indexOf('#') === 0 ? newDraftRef : ('#' + newDraftRef);

  calc.getRange(DOCNUM_OUT_CELL).setValue(excNum);

  // Tag the OLD order so the exchanged item is traceable to the new sale.
  const cleanOrderNum = orderNumber.replace('#', '');
  const orderData     = getOrderData(cleanOrderNum);
  if (orderData && orderData.id) {
    addOrderTags(orderData.id, [
      'exc-given',
      'exc-num:' + excNum,
      'exc-val:' + excValue.toFixed(2),
      'exc-applied-to:' + newDraftName,
      'exc-iss:' + issued
    ]);
  } else {
    ui.alert('⚠️ Old order ' + orderNumber + ' not found — Exchange Note applied but old order not tagged.');
  }

  log.appendRow([issued, excNum, orderNumber, newDraftName, customerName, customerEmail,
                 netWt, diaWt, goldVal, diaVal, excValue, 'Applied', String(newDraftId)]);

  var excCtx = afterSalesContext_(calc);
  logAfterSales_({
    'Logged At': issued, 'Event': 'issued',
    'Source': excCtx.Source, 'Exchange Type': excCtx['Exchange Type'], 'Doc Type': 'Exchange Note',
    'Store': excCtx.Store,
    'Serial': excNum, 'Order': orderNumber, 'Applied To': newDraftName,
    'Value': excValue, 'Net Wt': netWt, 'Dia Cts': diaWt, 'Gold Value': goldVal, 'Diamond Value': diaVal,
    'Customer': customerName, 'Email': customerEmail,
    // An exchange note is deducted on the spot, so it is 'applied' the moment it is raised — it
    // becomes a true redemption only when that draft converts, which refreshLogStatuses picks up.
    'Status': 'applied',
    'Reason / Note': String(calc.getRange(OVERRIDE_REASON_CELL).getValue()).trim(),
    'Ref': String(newDraftId)
  });

  sendExcEmail_(customerName, customerEmail, excNum, excValue, orderNumber, newDraftName);

  ui.alert(
    '✅ Exchange Note Applied\n\n' +
    'Exchange Note: ' + excNum + '\n' +
    'Deducted: ₹' + excValue.toLocaleString('en-IN', { minimumFractionDigits: 2 }) + '\n' +
    'Applied to: ' + newDraftName + '\n\n' +
    'The new invoice total is reduced by this amount (GST unchanged). ' +
    (SEND_CUSTOMER_EMAILS ? 'Email sent to ' + customerEmail + '.' : '✉️ Email SUPPRESSED (test mode).')
  );
}

// Applies the Exchange Note deduction to a new draft via the middleware. Returns the parsed
// JSON ({ success, draftId, ... }) or an { success:false, error } object on failure.
function applyExchangeNote_(newDraftRef, excNum, excValue, oldOrder, customerName) {
  try {
    var res = UrlFetchApp.fetch(MIDDLEWARE_URL + '/api/exc-redeem', {
      method:             'post',
      contentType:        'application/json',
      muteHttpExceptions: true,
      payload:            JSON.stringify({
        newDraftRef:     newDraftRef,
        excNumber:       excNum,
        excValue:        excValue,
        oldOrderNumber:  oldOrder,
        customerName:    customerName
      })
    });

    try { body = JSON.parse(res.getContentText()); } catch (e) {}

    }
    return body;
  } catch (e) {
    return { success: false, error: e.message };
  }
}

// Applies a Voucher to a new sale's draft via the middleware — the voucher twin of applyExchangeNote_.

// Shared tail for both voucher flows: if NEWDRAFT_CELL holds a draft ref, apply the freshly issued
// voucher to it. Returns { status, message, error } for the Voucher Log's Status column and the
// closing alert. A failure here NEVER discards the voucher — it exists in Shopify and the ledger
// either way, so we surface the reason and leave it open to apply from the admin app.
function applyVoucherIfDraftGiven_(calc, vchNumber, vchValue, oldOrder, customerName) {
  var ref = String(calc.getRange(NEWDRAFT_CELL).getValue()).trim();
  if (!ref) {
    return { status: 'Issued', error: '',
             message: 'Not applied to any draft — it is store credit the customer can spend later.\n\n' };
  }
  var name = ref.charAt(0) === '#' ? ref : ('#' + ref);
  var out  = applyVoucherToDraft_(ref, vchNumber, vchValue, oldOrder, customerName);
  if (out && out.success) {
    // Report what the middleware actually did. onlineCodeKilled is false when the ledger row carries
    // no price_rule_id (or the delete failed) — claiming otherwise would hide a live double-spend risk.
    return { status: 'Issued + Applied ' + name, error: '',
             message: (out.alreadyApplied ? 'Already applied to ' : 'Applied to ') + name + ' — ₹' +
                      Number(vchValue).toLocaleString('en-IN', { minimumFractionDigits: 2 }) +
                      ' deducted after tax.\n' +
                      (out.onlineCodeKilled
                         ? 'Its online discount code was cancelled — it cannot also be used at checkout.\n\n'
                         : '⚠️ Its online discount code is STILL LIVE — cancel ' + vchNumber +
                           ' under Discounts if the customer could also use it online.\n\n') };
  }
  var why = (out && out.error) || 'middleware error';
  return { status: 'Issued (apply failed)', error: why,
           message: 'Could NOT apply it to ' + name + ':\n  ' + why + '\n\n' +
                    'The voucher is valid and still open — check the draft number and apply it from the\n' +
                    'Shopify admin app (Adjustments → Voucher → paste ' + vchNumber + ').\n\n' };
}

// ── SERIAL — central counter via middleware ───────────────────────────────────

function allocateSerial_(docType, storeCode) {
  try {
    var res = UrlFetchApp.fetch(MIDDLEWARE_URL + '/api/serial/allocate', {
      method:             'post',
      contentType:        'application/json',
      muteHttpExceptions: true,
      // stateCode is REQUIRED for voucher/exchange_note serials (per-store); server 400s without it.
      payload:            JSON.stringify({ docType: docType, stateCode: storeCode || STORE_CODE })
    });
    if (res.getResponseCode() !== 200) {
      Logger.log(docType + ' serial warning: middleware returned ' + res.getResponseCode() + ' — ' + res.getContentText());
      return null;
    }
    var body = JSON.parse(res.getContentText());
    return (body && body.serial_no != null) ? body : null;
  } catch (e) {
    Logger.log(docType + ' serial failed: ' + e.message);
    return null;
  }
}

// ── VOID VOUCHER ──────────────────────────────────────────────────────────────
// A voucher can only be VOIDED before its expiry date. Voiding deletes the Shopify discount
// (price rule) and retires the serial in the ledger.
function voidVoucher() {
  var ss  = SpreadsheetApp.getActiveSpreadsheet();
  var log = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
  var ui  = SpreadsheetApp.getUi();
  if (!log) { ui.alert('Log tab "' + VOUCHER_LOG + '" not found.'); return; }

  var resp = ui.prompt('Void Voucher', 'Enter the voucher number to void (e.g. VCH-2026-0042):', ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;
  var cnNum = String(resp.getResponseText()).trim();
  if (!cnNum) { ui.alert('No voucher number entered.'); return; }

  // Locate the voucher in the log (col B = number).
  var data = log.getDataRange().getValues();
  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][1]).trim().toUpperCase() === cnNum.toUpperCase()) { rowIdx = i; break; }
  }
  if (rowIdx === -1) { ui.alert(cnNum + ' not found in ' + VOUCHER_LOG + '.'); return; }

  var row         = data[rowIdx];
  var expiryFmt   = String(row[10]).trim();  // col K — dd-MM-yyyy
  var status      = String(row[11]).trim();  // col L
  var priceRuleId = String(row[12]).trim();  // col M

  if (/void/i.test(status)) { ui.alert(cnNum + ' is already voided.'); return; }

  // Only voidable before expiry.
  var expDate = parseDmy(expiryFmt);
  if (expDate && new Date() > expDate) {
    ui.alert(cnNum + ' expired on ' + expiryFmt + ' — it can no longer be voided.');
    return;
  }

  var confirm = ui.alert('Void ' + cnNum + '?',
    'This deletes the Shopify discount and retires the serial. This cannot be undone.',
    ui.ButtonSet.YES_NO);
  if (confirm !== ui.Button.YES) return;

  // 1. Delete the Shopify price rule (removes its discount code too).
  if (priceRuleId) {
    shopifyDelete('price_rules/' + priceRuleId + '.json');
  } else {
    Logger.log('Void ' + cnNum + ': no price_rule_id on log row — skipping Shopify delete.');
  }

  // 2. Retire the serial (by seq parsed from the voucher number).
  var seq = parseInt(String(cnNum).split('-').pop(), 10);
  var retired = cancelSerialByCode_('voucher', seq);

  // 3. Mark the log row voided (col L).
  log.getRange(rowIdx + 1, 12).setValue('Voided');

  // A void is an event in its own right, not just a status flip — the report has to be able to show
  // that value was issued and then withdrawn, with the date it happened.
  logAfterSales_({
    'Event': 'voided', 'Doc Type': 'Voucher', 'Serial': cnNum,
    'Order': String(row[2]).trim(), 'Value': row[9],
    'Customer': String(row[3]).trim(), 'Email': String(row[4]).trim(),
    'Expiry': expiryFmt, 'Status': 'voided',
    'Reason / Note': retired ? 'Discount deleted, serial retired' : 'Discount deleted, LEDGER RETIRE FAILED',
    'Ref': 'void:' + cnNum
  });

  ui.alert('✅ ' + cnNum + ' voided.\n\nDiscount removed' +
           (retired ? ' and serial retired in the ledger.' : '. ⚠️ Ledger retire failed — check the logs.'));
}

// ── VOID EXCHANGE NOTE ────────────────────────────────────────────────────────
// Removes the EXC line item from the new draft and retires the serial. Only possible while the
// new sale is still a DRAFT (the middleware refuses if it has converted to an order).
function voidExchangeNote() {
  var ss  = SpreadsheetApp.getActiveSpreadsheet();
  var log = ss.getSheetByName(EXCHANGE_LOG);
  var ui  = SpreadsheetApp.getUi();
  if (!log) { ui.alert('Log tab "' + EXCHANGE_LOG + '" not found.'); return; }

  var resp = ui.prompt('Void Exchange Note', 'Enter the EXC number to void (e.g. EXC-2026-0042):', ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;
  var excNum = String(resp.getResponseText()).trim();
  if (!excNum) { ui.alert('No EXC number entered.'); return; }

  var data = log.getDataRange().getValues();
  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][1]).trim().toUpperCase() === excNum.toUpperCase()) { rowIdx = i; break; }
  }
  if (rowIdx === -1) { ui.alert(excNum + ' not found in ' + EXCHANGE_LOG + '.'); return; }

  var row        = data[rowIdx];
  var oldOrder   = String(row[2]).trim();   // col C
  var status     = String(row[11]).trim();  // col L
  var newDraftId = String(row[12]).trim();  // col M — resolved numeric draft id

  if (/void/i.test(status)) { ui.alert(excNum + ' is already voided.'); return; }

  var confirm = ui.alert('Void ' + excNum + '?',
    'This removes the exchange line from the new draft and retires the serial. Only works if the new sale is still a draft.',
    ui.ButtonSet.YES_NO);
  if (confirm !== ui.Button.YES) return;

  // Ask the middleware to remove the EXC line + cancel the serial.
  try {
    var res = UrlFetchApp.fetch(MIDDLEWARE_URL + '/api/exc-void', {
      method:             'post',
      contentType:        'application/json',
      muteHttpExceptions: true,
      payload:            JSON.stringify({ newDraftId: newDraftId, excNumber: excNum })
    });
    var body = {};
    try { body = JSON.parse(res.getContentText()); } catch (e) {}
    if (res.getResponseCode() !== 200) {
      ui.alert('❌ Could not void ' + excNum + ':\n\n' + ((body && body.error) || ('middleware ' + res.getResponseCode())));
      return;
    }
  } catch (e) {
    ui.alert('❌ Void failed: ' + e.message);
    return;
  }

  // Strip the exc-* tags from the old order.
  if (oldOrder) {
    var od = getOrderData(oldOrder.replace('#', ''));
    if (od && od.id) removeOrderTagsByPrefix_(od.id, ['exc-given', 'exc-num:', 'exc-val:', 'exc-applied-to:', 'exc-iss:']);
  }

  log.getRange(rowIdx + 1, 12).setValue('Voided');

  logAfterSales_({
    'Event': 'voided', 'Doc Type': 'Exchange Note', 'Serial': excNum,
    'Order': oldOrder, 'Applied To': String(row[3]).trim(), 'Value': row[10],
    'Customer': String(row[4]).trim(), 'Email': String(row[5]).trim(),
    'Status': 'voided', 'Reason / Note': 'Exchange line removed from the draft, serial retired',
    'Ref': 'void:' + excNum
  });

  ui.alert('✅ ' + excNum + ' voided.\n\nExchange line removed from the draft and serial retired.');
}

// Removes any tags on an order that exactly match or start with one of the given prefixes.
function removeOrderTagsByPrefix_(orderId, prefixes) {
  var data = shopifyGet('orders/' + orderId + '.json?fields=id,tags');
  if (!data || !data.order) return;
  var existing = data.order.tags ? data.order.tags.split(', ').map(function (t) { return t.trim(); }) : [];
  var kept = existing.filter(function (t) {
    return !prefixes.some(function (p) { return t === p || t.indexOf(p) === 0; });
  });
  shopifyPut('orders/' + orderId + '.json', { order: { id: orderId, tags: kept.join(', ') } });
}

// Parses a dd-MM-yyyy string into a Date (end-of-day, IST-agnostic). Returns null if unparseable.
function parseDmy(s) {
  var m = /^(\d{1,2})-(\d{1,2})-(\d{4})$/.exec(String(s).trim());
  if (!m) return null;
  return new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]), 23, 59, 59);
}

// ── HELPERS ───────────────────────────────────────────────────────────────────
function toNum(val) {
  if (typeof val === 'number') return val;
  return parseFloat(String(val).replace(/[^0-9.-]/g, '')) || 0;
}

// ── TOKEN ─────────────────────────────────────────────────────────────────────
function getToken() {
  const props       = PropertiesService.getScriptProperties();
  const supabaseUrl = props.getProperty('SUPABASE_URL');
  const supabaseKey = props.getProperty('SUPABASE_SERVICE_KEY');

  if (!supabaseUrl || !supabaseKey) {
    throw new Error('Supabase credentials not set. Use "Setup Supabase Credentials" from the menu.');
  }

  const res = UrlFetchApp.fetch(
    supabaseUrl + '/rest/v1/config?key=eq.shopify_access_token&select=value',
    {
      headers: { 'apikey': supabaseKey, 'Authorization': 'Bearer ' + supabaseKey },
      muteHttpExceptions: true
    }
  );

  if (res.getResponseCode() >= 400) {
    throw new Error('Supabase token fetch failed: ' + res.getContentText());
  }

  const rows = JSON.parse(res.getContentText());
  if (!rows || rows.length === 0) throw new Error('shopify_access_token not found in Supabase config table');
  return rows[0].value;
}

// Reads any row from the Supabase `config` table by key. Returns the raw string value, or null if
// absent. Same credentials/path as getToken (the token is just the shopify_access_token key).
function getSupabaseConfig_(key) {
  const props       = PropertiesService.getScriptProperties();
  const supabaseUrl = props.getProperty('SUPABASE_URL');
  const supabaseKey = props.getProperty('SUPABASE_SERVICE_KEY');
  if (!supabaseUrl || !supabaseKey) throw new Error('Supabase credentials not set. Use "Setup Supabase Credentials".');
  const res = UrlFetchApp.fetch(
    supabaseUrl + '/rest/v1/config?key=eq.' + encodeURIComponent(key) + '&select=value',
    { headers: { 'apikey': supabaseKey, 'Authorization': 'Bearer ' + supabaseKey }, muteHttpExceptions: true });
  if (res.getResponseCode() >= 400) throw new Error('Supabase config fetch failed: ' + res.getContentText());
  const rows = JSON.parse(res.getContentText());
  return (rows && rows.length) ? rows[0].value : null;
}

// Old-gold buy-back rate per gram for a (possibly fractional) karat. Reads the SAME buying_rate_table
// the middleware uses (server.js getBuyingRateTable/buyingRateFor), so the sheet and any middleware
// re-valuation agree to the paise. Formula: karat/24 × base_24k × (1 − haircut_pct/100).
// Returns null if the table is missing or purity is outside 0<p<=24.
function getBuyingRate_(purity) {
  var p = toNum(purity);
  if (!(p > 0) || p > 24) return null;
  var raw = getSupabaseConfig_('buying_rate_table');
  if (!raw) return null;
  var t;
  try { t = JSON.parse(raw); } catch (e) { return null; }
  if (!t || !(t.base_24k > 0)) return null;
  return Math.round((p / 24) * t.base_24k * (1 - (t.haircut_pct || 0) / 100) * 100) / 100;
}

// Looks up a customer by phone or email via the Shopify Admin API. Returns { id, name, email, phone }
// or null. Used by the Old Gold flow, which has no order to pull a customer from.
function lookupCustomer_(query) {
  var q = String(query || '').trim();
  if (!q) return null;
  var field = q.indexOf('@') !== -1 ? 'email' : 'phone';
  var data = shopifyGet('customers/search.json?query=' + encodeURIComponent(field + ':' + q));
  if (!data || !data.customers || !data.customers.length) return null;
  var c = data.customers[0];
  return {
    id:    c.id,
    name:  [c.first_name, c.last_name].filter(Boolean).join(' '),
    email: c.email || '',
    phone: (c.phone || (c.default_address && c.default_address.phone) || '')
  };
}

// Reads the source mode (Purchase Exchange | Old Gold) from SOURCE_CELL, lower-cased. Blank = default.
function currentSource_(calc) {
  return String(calc.getRange(SOURCE_CELL).getValue()).trim().toLowerCase();
}

// Keeps FINAL_ISSUE_CELL ("Value to Issue") in sync with the chosen mode — this is what the create
// functions issue against, and what staff see change when they flip the dropdowns:
//   Old Gold                 → weight × buy-back rate. No order, so the table is irrelevant.
//   Purchase Exchange, both  → =NET_VALUE_CELL. The mode is expressed in the TABLE now, not here.
// Never writes NET_VALUE_CELL — that stays a formula.
// THIS CELL IS THE ONLY ONE THAT ISSUES MONEY. createVoucher_ and createExchangeNote_ each read
// FINAL_ISSUE_CELL and nothing else; NET_VALUE_CELL and the column total are never read by any code.
//
// Full Value used to bypass the table and write the order's Shopify subtotal straight in. Three
// things were wrong with that, and moving the mode into refreshExchangeColumn_ fixes all three:
//
//   • the subtotal is an ORDER-level figure, so exchanging ONE line of a multi-item order credited
//     every item on it — while the SKU picker exists precisely to exchange one;
//   • on a tax-inclusive store the subtotal IS the gross, so it was issued under a "taxable" label;
//   • the issue cell diverging from the table above it was invisible, and read as a bug every time.
//
// Now both Purchase Exchange modes mirror NET, and the before/after table shows how that NET was
// reached. The Why column beside the cell still names the mode that produced the number.
function refreshFinalIssueValue_(calc) {
  var cell = calc.getRange(FINAL_ISSUE_CELL);
  var why  = calc.getRange(cellRow_(FINAL_ISSUE_CELL), WHY_COL);
  var explain = function (text) {
    why.setValue(text).setFontColor('#777777').setFontStyle('italic');
  };

  if (currentSource_(calc) === SOURCE_OLD_GOLD) {
    cell.setValue(oldGoldValue_(calc));
    cell.setNote('Old Gold: weight × buy-back rate. Does not read the table above.');
    explain('Old Gold — weight × buy-back rate (table above not used)');
    return;
  }
  // Write the table for the selected mode first — the issue cell is only ever its NET.
  refreshExchangeColumn_(calc);
  cell.setFormula('=' + NET_VALUE_CELL);
  cell.clearNote();
  explain(isFullValue_(calc)
    ? 'Full Value — the invoice as charged, tax inclusive (mirrors NET)'
    : 'Deduction — gold + diamond revalued today (mirrors NET)');
}

// Full Value credits the tax-inclusive amount, so a blank GST row silently issues the TAXABLE value
// instead — a plausible-looking number that is short by exactly the tax. Returns false when the
// document must not be issued.
function fullValueGuardPasses_(calc, ui, orderNumber, value) {
  if (!isFullValue_(calc)) return true;
  if (!(value > 0)) {
    ui.alert('Full Value selected but the exchange column for ' + orderNumber + ' is empty.\n\n' +
             'Re-enter the order number in ' + ORDER_NUM_CELL + ' to repopulate it, or switch to Deduction.');
    return false;
  }
  if (String(calc.getRange(PAID_TAX_CELL).getValue()).trim() !== '') return true;
  return ui.alert('GST not recorded on ' + orderNumber,
    'Full Value credits the entire tax-inclusive amount, but this order recorded no GST figure, so ' +
    PAID_TAX_CELL + ' is blank and the credit stops at the taxable value.\n\n' +
    'Issue ' + value.toFixed(2) + ' anyway?', ui.ButtonSet.YES_NO) === ui.Button.YES;
}

function getOrderData(orderName) {
  const data = shopifyGet('orders.json?name=%23' + orderName + '&fields=id,tags,customer&status=any');
  if (data && data.orders && data.orders.length > 0) {
    const order = data.orders[0];
    return { id: order.id, customerId: order.customer ? order.customer.id : null };
  }
  return null;
}

function addOrderTags(orderId, newTags) {
  const data = shopifyGet('orders/' + orderId + '.json?fields=id,tags');
  if (!data) return;
  const existing = data.order.tags ? data.order.tags.split(', ').map(function(t) { return t.trim(); }) : [];
  newTags.forEach(function(tag) {
    // Remove any existing tag with the same prefix (e.g. old cn-val:) before adding the new one
    var prefix = tag.indexOf(':') !== -1 ? tag.split(':')[0] + ':' : null;
    if (prefix) {
      var idx = existing.findIndex(function(t) { return t.indexOf(prefix) === 0; });
      if (idx !== -1) existing.splice(idx, 1);
    }
    if (!existing.includes(tag)) existing.push(tag);
  });
  shopifyPut('orders/' + orderId + '.json', { order: { id: orderId, tags: existing.join(', ') } });
}

    const body = res.getContentText();
    if (code >= 400) { ui.alert('❌ Shopify returned ' + code + ':\n' + body.substring(0, 300)); return; }
    const data = JSON.parse(body);
    if (data && data.shop) {
      ui.alert('✅ Connected to: ' + data.shop.name + '\nDomain: ' + data.shop.domain);
    } else {
      ui.alert('❌ Unexpected response:\n' + body.substring(0, 300));
    }
  } catch (err) {
    ui.alert('❌ Error: ' + err.message);
  }
}


// ═════════════════════════════════════════════════════════════════════════════
// INBOUND WEB APP — issue_voucher
//
// Lets the middleware raise a voucher in this sheet when a customer clicks
// "Refund to my Timanti wallet" on the repair ready+final email.
//
// The sheet stays the system of record: the middleware supplies only what it
// alone knows (store code, refund amount, customer, source order) and this
// issues the voucher exactly as the counter flow does — same serial series,
// same Shopify discount code, same ledger row, same customer email.
//
// DEPLOY: Deploy → New deployment → Web app. Execute as: Me. Access: Anyone.
// Then set that URL on Fly as EXCHANGE_APPS_SCRIPT_URL (NOT APPS_SCRIPT_URL,
// which is the PO Tracker and would swallow the request silently).
// On later edits, deploy to the SAME deployment so the URL does not change.
// ═════════════════════════════════════════════════════════════════════════════

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  if (body.action === 'issue_voucher')   return handleIssueVoucher_(body);
  if (body.action === 'log_cad_advance') return handleCadAdvanceLog_(body);
  return ContentService
    .createTextOutput(JSON.stringify({ ok: false, error: 'unknown action' }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ═════════════════════════════════════════════════════════════════════════════
// CAD ADVANCE LOG — inbound from the middleware, one row per state change
//
// Posted on every state change of a design advance:
//
//   { action:'log_cad_advance',
//     event:'captured'|'applied'|'redeemed'|'expired'|'deleted'|'refunded',
//     draft_ref:'#D205', value:5000, customer_name:'A Kumar',
//     expires_at:'2027-09-01', against:'#1042', draft_id:'1461677883649',
//     logged_at:'2026-09-07T13:00:00.000Z' }
//
// This is where STORE STAFF see advances. The Supabase register holds current state and reaches
// accounts by email once a month; neither is somewhere the counter looks, and Shopify shows one
// document at a time.
//
// APPEND ONLY — one row per event, never an update in place. An advance taken, applied and then
// redeemed reads as three lines, so staff can see when each thing happened.
//
// The tab and header row are created on first use; nothing to set up by hand.
// ═════════════════════════════════════════════════════════════════════════════

var CAD_ADVANCE_LOG = 'CAD Advance Log';

function handleCadAdvanceLog_(body) {
  var out = function (ok, payload) {
    var o = { ok: ok };
    for (var k in (payload || {})) o[k] = payload[k];
    return ContentService.createTextOutput(JSON.stringify(o))
      .setMimeType(ContentService.MimeType.JSON);
  };

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ss.getSheetByName(CAD_ADVANCE_LOG);

    if (!sh) {
      sh = ss.insertSheet(CAD_ADVANCE_LOG);
      sh.appendRow(['Logged At (IST)', 'Event', 'Reference', 'Value',
                    'Customer', 'Expires On', 'Applied / Redeemed Against', 'Draft ID']);
      sh.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#f0f0f0');
      sh.setFrozenRows(1);
    }

    // Times arrive as UTC ISO from the server; staff read IST.
    var when = body.logged_at ? new Date(body.logged_at) : new Date();
    var whenIst = Utilities.formatDate(when, 'Asia/Kolkata', 'yyyy-MM-dd HH:mm');

    sh.appendRow([
      whenIst,
      String(body.event || ''),
      String(body.draft_ref || ''),
      (body.value === '' || body.value == null) ? '' : Number(body.value),
      String(body.customer_name || ''),
      body.expires_at ? String(body.expires_at).slice(0, 10) : '',
      String(body.against || ''),
      String(body.draft_id || '')
    ]);

    return out(true, { logged: body.event + ' ' + body.draft_ref });
  } catch (err) {
    // The middleware ignores the body and only cares that the POST did not error, but returning the
    // message makes a broken sheet visible in its logs instead of silently losing rows.
    Logger.log('log_cad_advance failed: ' + err);
    return out(false, { error: String(err) });
  }
}

/** Run once from the editor to prove the tab and the append work, before any real advance exists. */
function testCadAdvanceLog_() {
  Logger.log(handleCadAdvanceLog_({
    event: 'captured', draft_ref: '#TEST-DELETE-ME', value: 5000,
    customer_name: 'Test row — safe to delete', expires_at: '2027-09-01',
    logged_at: new Date().toISOString()
  }).getContent());
}

function handleIssueVoucher_(body) {
  var out = function (ok, payload) {
    return ContentService
      .createTextOutput(JSON.stringify(Object.assign({ ok: ok }, payload || {})))
      .setMimeType(ContentService.MimeType.JSON);
  };

  try {
    var value = Number(body.value);
    if (!body.store_code)     return out(false, { error: 'store_code is required' });
    if (!(value > 0))         return out(false, { error: 'value must be > 0' });
    if (!body.customer_email) return out(false, { error: 'customer_email is required' });

    var ss  = SpreadsheetApp.getActiveSpreadsheet();
    var log = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
    if (!log) return out(false, { error: 'Voucher Log sheet not found' });

    // Idempotency — a retried POST returns the original, never issues twice.
    var existing = findVoucherBySourceRef_(log, body.draft_ref);
    if (existing) return out(true, { cnNumber: existing.cnNumber, duplicate: true });

    // Same central allocator the counter flow uses, so the per-store sequence stays continuous.
    // Store code comes from the repair draft, not the sheet's control cell.
    var alloc = allocateSerial_('voucher', body.store_code);
    if (!alloc || !alloc.serial_code) {
      return out(false, { error: 'could not allocate a voucher serial - middleware unreachable' });
    }
    var cnNumber = alloc.serial_code;

    var tz         = ss.getSpreadsheetTimeZone();
    var issued     = new Date();
    var expires    = new Date(issued.getFullYear() + 1, issued.getMonth(), issued.getDate());
    var issuedStr  = Utilities.formatDate(issued,  tz, 'dd-MM-yyyy');
    var expiresStr = Utilities.formatDate(expires, tz, 'dd-MM-yyyy');

    // Bind the discount to the customer, or the voucher becomes a bearer instrument
    // redeemable by anyone who sees the code.
    var cust   = lookupCustomer_(body.customer_email);
    var custId = cust ? cust.id : null;

    // A voucher IS a Shopify discount code. Without the price rule the customer
    // receives a number that buys nothing.
    var prPayload = {
      title:              cnNumber,
      target_type:        'line_item',
      target_selection:   'all',
      allocation_method:  'across',
      value_type:         'fixed_amount',
      value:              '-' + value.toFixed(2),
      customer_selection: custId ? 'prerequisite' : 'all',
      starts_at:          issued.toISOString(),
      ends_at:            expires.toISOString(),
      usage_limit:        1
    };
    if (custId) prPayload.prerequisite_customer_ids = [Number(custId)];

    var priceRule = shopifyPost('price_rules.json', { price_rule: prPayload });
    if (!priceRule || !priceRule.price_rule) return out(false, { error: 'Shopify price rule creation failed' });
    var priceRuleId = priceRule.price_rule.id;

    var discCode = shopifyPost('price_rules/' + priceRuleId + '/discount_codes.json',
                               { discount_code: { code: cnNumber } });
    if (!discCode || !discCode.discount_code) return out(false, { error: 'Shopify discount code creation failed' });

    // Ledger row. Without it handleApplyVoucherTag finds nothing and tags the draft
    // "voucher-invalid: not found", deducting zero.
    var ledgerOk = issueCreditInstrument_({
      instrumentType:  'voucher',
      serialCode:      cnNumber,
      value:           value,
      customerId:      custId ? String(custId) : null,
      customerName:    body.customer_name,
      sourceOrderName: body.source_order || body.draft_ref,
      stateCode:       body.store_code,
      expiresAt:       expires.toISOString(),
      priceRuleId:     String(priceRuleId)
    });

    // Columns mirror a counter-raised voucher THROUGH column M (price_rule_id), because
    // voidVoucher reads that index. The repair-refund extras go after it, never in place of it.
    // Weight/valuation columns are blank on purpose - a repair refund has no gold/diamond
    // valuation, the amount is estimate minus actual.
    log.appendRow([
      issuedStr,                      // A  issued
      cnNumber,                       // B  voucher number
      body.source_order || '',        // C  original order
      body.customer_name || '',       // D  customer
      body.customer_email,            // E  email
      '',                             // F  net wt      - n/a
      '',                             // G  dia cts     - n/a
      '',                             // H  gold value  - n/a
      '',                             // I  dia value   - n/a
      value,                          // J  net credit
      expiresStr,                     // K  expiry
      'Issued',                       // L  status
      String(priceRuleId),            // M  price_rule_id  <- voidVoucher reads this
      body.source || 'repair-refund', // N  why it was raised
      body.draft_ref || ''            // O  repair draft - idempotency key
    ]);

    // Headless path - there are no control cells to read, so the classification is fixed here.
    // Store code comes off the repair draft, not resolveStoreCode_(), for the same reason.
    logAfterSales_({
      'Logged At': issuedStr, 'Event': 'issued',
      'Source': 'Repair refund', 'Exchange Type': '—', 'Doc Type': 'Voucher', 'Store': body.store_code,
      'Serial': cnNumber, 'Order': body.source_order || '', 'Applied To': body.draft_ref || '',
      'Value': value,
      'Customer': body.customer_name || '', 'Email': body.customer_email, 'Expiry': expiresStr,
      'Status': 'outstanding',
      'Reason / Note': (body.source || 'repair-refund') +
                       (ledgerOk ? '' : ' — LEDGER WRITE FAILED, this voucher will not apply'),
      'Ref': body.draft_ref || String(priceRuleId)
    });

    // Tag the original order so a refund-raised voucher looks like any other.
    if (body.source_order) {
      try {
        var od = getOrderData(String(body.source_order).replace('#', ''));
        if (od && od.id) {
          addOrderTags(od.id, [
            'cn-issued',
            'cn-num:' + cnNumber,
            'cn-val:' + value.toFixed(2),
            'cn-exp:' + expiresStr,
            'cn-iss:' + issuedStr
          ]);
        }
      } catch (tagErr) {
        Logger.log('issue_voucher: order tagging failed for ' + body.source_order + ': ' + tagErr);
      }
    }

    // Reuse the counter flow's sender, so the email is identical and the
    // SEND_CUSTOMER_EMAILS kill-switch governs this path too.
    sendVoucherEmail_(body.customer_name, body.customer_email, cnNumber, value,
                      expiresStr, body.source_order || '');

    return out(true, { cnNumber: cnNumber, value: value, expires: expiresStr, ledger: ledgerOk });

  } catch (err) {
    Logger.log('issue_voucher failed: ' + err);
    return out(false, { error: String(err) });
  }
}

// Finds an existing voucher by repair draft reference so a retry returns the original.
// draft_ref is column O (index 14) in the row appended above.
function findVoucherBySourceRef_(log, draftRef) {
  if (!draftRef) return null;
  var values = log.getDataRange().getValues();
  for (var i = values.length - 1; i >= 1; i--) {
    if (String(values[i][14] || '').trim() === String(draftRef).trim()) {
      return { cnNumber: values[i][1], row: i + 1 };
    }
  }
  return null;
}

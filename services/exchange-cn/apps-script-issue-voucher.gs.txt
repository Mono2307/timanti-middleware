/**
 * issue_voucher — Apps Script handler for vouchers raised from an email CTA.
 *
 * The middleware calls this when a customer clicks "Refund to my Timanti wallet"
 * on the repair ready+final email.
 *
 * WHY THE SHEET AND NOT THE MIDDLEWARE
 * The sheet is the system of record for credit instruments — the Voucher Log is
 * what finance reconciles against. A voucher minted only in the middleware would
 * be invisible there. So the middleware supplies the facts it alone knows (store
 * code, delta, customer, source order) and the sheet issues it exactly as it does
 * for a counter-raised voucher.
 *
 * WHAT THE MIDDLEWARE SENDS
 * {
 *   action:         'issue_voucher',
 *   source:         'repair-refund',
 *   draft_ref:      '#D1042',
 *   store_code:     'KA-HSR',      // custom.state_code on the repair draft
 *   value:          870,           // collected minus final cost
 *   customer_name:  'Priya Sharma',
 *   customer_email: 'priya@example.com',
 *   source_order:   '#1042'        // custom.repair_order_reference
 * }
 *
 * ─────────────────────────────────────────────────────────────────────────────
 * WHERE THIS GOES — the Exchange Calculator's Apps Script project, NOT the PO one
 *
 *   1. Open the Exchange Calculator sheet → Extensions → Apps Script.
 *   2. Paste everything below the divider at the END of the existing script.
 *      Do NOT replace the file — this reuses its constants and functions.
 *   3. The project has no doPost today, so add the router below too.
 *   4. Deploy → New deployment → Web app. Execute as: Me. Access: Anyone.
 *   5. Set the URL on Fly as EXCHANGE_APPS_SCRIPT_URL:
 *        flyctl secrets set EXCHANGE_APPS_SCRIPT_URL="https://script.google.com/..."
 *
 * DO NOT reuse APPS_SCRIPT_URL — that is the PO Tracker web app, which handles
 * only 'append' and 'update'. An issue_voucher post there is silently ignored.
 *
 * Re-deploy to the SAME deployment on later edits so the URL does not change.
 *
 * DEPENDS ON (all already in the script — do not redefine):
 *   VOUCHER_LOG, MIDDLEWARE_URL, SEND_CUSTOMER_EMAILS, allocateSerial_,
 *   issueCreditInstrument_, sendVoucherEmail_, lookupCustomer_, getOrderData,
 *   addOrderTags, shopifyPost
 * ─────────────────────────────────────────────────────────────────────────────
 */

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  if (body.action === 'issue_voucher') return handleIssueVoucher_(body);
  return ContentService
    .createTextOutput(JSON.stringify({ ok: false, error: 'unknown action' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function handleIssueVoucher_(body) {
  var out = function (ok, payload) {
    return ContentService
      .createTextOutput(JSON.stringify(Object.assign({ ok: ok }, payload || {})))
      .setMimeType(ContentService.MimeType.JSON);
  };

  try {
    var value = Number(body.value);
    if (!body.store_code)     return out(false, { error: 'store_code is required' });
    if (!(value > 0))         return out(false, { error: 'value must be > 0' });
    if (!body.customer_email) return out(false, { error: 'customer_email is required' });

    var ss  = SpreadsheetApp.getActiveSpreadsheet();
    var log = ss.getSheetByName(VOUCHER_LOG) || ss.getSheetByName('CN Log');
    if (!log) return out(false, { error: 'Voucher Log sheet not found' });

    // Idempotency — a retried POST must return the original, never issue twice.
    var existing = findVoucherBySourceRef_(log, body.draft_ref);
    if (existing) return out(true, { cnNumber: existing.cnNumber, duplicate: true });

    // Serial from the SAME central allocator the counter flow uses, so the sequence stays
    // continuous per store. Store code comes from the repair draft, not the sheet's control cell.
    var alloc = allocateSerial_('voucher', body.store_code);
    if (!alloc || !alloc.serial_code) {
      return out(false, { error: 'could not allocate a voucher serial — middleware unreachable' });
    }
    var cnNumber = alloc.serial_code;

    var tz         = ss.getSpreadsheetTimeZone();
    var issued     = new Date();
    var expires    = new Date(issued.getFullYear() + 1, issued.getMonth(), issued.getDate());
    var issuedStr  = Utilities.formatDate(issued,  tz, 'dd-MM-yyyy');
    var expiresStr = Utilities.formatDate(expires, tz, 'dd-MM-yyyy');

    // Resolve the customer so the discount is bound to them. Without this the voucher would be a
    // bearer instrument redeemable by anyone who saw the code.
    var cust   = lookupCustomer_(body.customer_email);
    var custId = cust ? cust.id : null;

    // A voucher IS a Shopify discount code — without the price rule the customer receives a number
    // that buys nothing. Same shape as createVoucher_.
    var prPayload = {
      title:              cnNumber,
      target_type:        'line_item',
      target_selection:   'all',
      allocation_method:  'across',
      value_type:         'fixed_amount',
      value:              '-' + value.toFixed(2),
      customer_selection: custId ? 'prerequisite' : 'all',
      starts_at:          issued.toISOString(),
      ends_at:            expires.toISOString(),
      usage_limit:        1
    };
    if (custId) prPayload.prerequisite_customer_ids = [Number(custId)];

    var priceRule = shopifyPost('price_rules.json', { price_rule: prPayload });
    if (!priceRule || !priceRule.price_rule) return out(false, { error: 'Shopify price rule creation failed' });
    var priceRuleId = priceRule.price_rule.id;

    var discCode = shopifyPost('price_rules/' + priceRuleId + '/discount_codes.json',
                               { discount_code: { code: cnNumber } });
    if (!discCode || !discCode.discount_code) return out(false, { error: 'Shopify discount code creation failed' });

    // Ledger row — without it handleApplyVoucherTag finds nothing and tags the draft
    // "voucher-invalid: not found", deducting zero.
    var ledgerOk = issueCreditInstrument_({
      instrumentType:  'voucher',
      serialCode:      cnNumber,
      value:           value,
      customerId:      custId ? String(custId) : null,
      customerName:    body.customer_name,
      sourceOrderName: body.source_order || body.draft_ref,
      stateCode:       body.store_code,
      expiresAt:       expires.toISOString(),
      priceRuleId:     String(priceRuleId)
    });

    // Column order MIRRORS a counter-raised voucher through column M (price_rule_id) — voidVoucher
    // reads that index, so the extra repair-refund columns go AFTER it, never in place of it.
    // Weight/valuation columns are blank on purpose: a repair refund has no gold/diamond valuation,
    // the amount is estimate-minus-actual.
    log.appendRow([
      issuedStr,                      // A  issued
      cnNumber,                       // B  voucher number
      body.source_order || '',        // C  original order
      body.customer_name || '',       // D  customer
      body.customer_email,            // E  email
      '',                             // F  net wt      — n/a
      '',                             // G  dia cts     — n/a
      '',                             // H  gold value  — n/a
      '',                             // I  dia value   — n/a
      value,                          // J  net credit
      expiresStr,                     // K  expiry
      'Issued',                       // L  status
      String(priceRuleId),            // M  price_rule_id  ← voidVoucher reads this
      body.source || 'repair-refund', // N  why it was raised
      body.draft_ref || ''            // O  repair draft — idempotency key
    ]);

    // Tag the original order so a refund-raised voucher looks like any other.
    if (body.source_order) {
      try {
        var od = getOrderData(String(body.source_order).replace('#', ''));
        if (od && od.id) {
          addOrderTags(od.id, [
            'cn-issued',
            'cn-num:' + cnNumber,
            'cn-val:' + value.toFixed(2),
            'cn-exp:' + expiresStr,
            'cn-iss:' + issuedStr
          ]);
        }
      } catch (tagErr) {
        Logger.log('issue_voucher: order tagging failed for ' + body.source_order + ': ' + tagErr);
      }
    }

    // Reuse the counter flow's sender so the customer email is identical, and so the
    // SEND_CUSTOMER_EMAILS kill-switch governs this path too.
    sendVoucherEmail_(body.customer_name, body.customer_email, cnNumber, value,
                      expiresStr, body.source_order || '');

    return out(true, { cnNumber: cnNumber, value: value, expires: expiresStr, ledger: ledgerOk });

  } catch (err) {
    Logger.log('issue_voucher failed: ' + err);
    return out(false, { error: String(err) });
  }
}

/**
 * Finds an existing voucher by repair draft reference so a retry returns the original.
 * draft_ref is column O (index 14) in the row appended above.
 */
function findVoucherBySourceRef_(log, draftRef) {
  if (!draftRef) return null;
  var values = log.getDataRange().getValues();
  for (var i = values.length - 1; i >= 1; i--) {
    if (String(values[i][14] || '').trim() === String(draftRef).trim()) {
      return { cnNumber: values[i][1], row: i + 1 };
    }
  }
  return null;
}
